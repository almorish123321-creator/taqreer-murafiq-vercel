   577→    function fmtTime12(dtStr) {
   578→        if (!dtStr) return "";
   579→        const t = dtStr.split('T')[1] || "00:00";
   580→        let [h, m] = t.split(':').map(Number);
   581→        const ampm = h >= 12 ? 'PM' : 'AM';
   582→        const ampmAr = h >= 12 ? 'مساءً' : 'صباحاً';
   583→        h = h % 12 || 12;
   584→        const hh = (h < 10 ? '0' + h : '' + h);
   585→        const mm = (m < 10 ? '0' + m : '' + m);
   586→        return { en: hh + ':' + mm + ' ' + ampm, ar: hh + ':' + mm + ' ' + ampmAr };
   587→    }

   590→    function waitArabic(dInVal, dOutVal) {
   591→        if (!dInVal || !dOutVal) return "";
   592→        const d1 = new Date(dInVal), d2 = new Date(dOutVal);
   593→        if (d2 < d1) return "";
   594→        const totalMin = Math.round((d2 - d1) / 60000);
   595→        const hrs = Math.floor(totalMin / 60);
   596→        const mins = totalMin % 60;
   597→        let parts = [];
   598→        if (hrs > 0) parts.push(hrs + " ساعة");
   599→        if (mins > 0) parts.push(mins + " دقيقة");
   600→        return parts.join(" و ") || "0 دقيقة";
   601→    }
