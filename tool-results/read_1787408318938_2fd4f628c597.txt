   411→    formatGregorian(dateString) {
   412→        if(!dateString) return "";
   413→        const parts = dateString.split('-');
   414→        if(parts.length===3) return `${parts[2]}-${parts[1]}-${parts[0]}`;
   415→        return dateString;

   418→    formatAMPM(timeStr) {
   419→        if(!timeStr) return "";
   420→        let [hours, minutes] = timeStr.split(':');
   421→        hours = parseInt(hours);
   422→        let ampm = hours >= 12 ? 'PM' : 'AM';
   423→        hours = hours % 12;
   424→        hours = hours ? hours : 12; 
   425→        return `${hours}:${minutes} ${ampm}`;
   426→    },

   885→    printLastPdf() {
   886→        if (!app.state.lastPdfBase64) {
   887→            if(this.tg) this.tg.showAlert('لا يوجد تقرير للطباعة');
   888→            else alert('لا يوجد تقرير للطباعة');
   889→            return;
   890→        }
   891→        try {
   892→            const rawBase64 = app.state.lastPdfBase64.split(',')[1] || app.state.lastPdfBase64;
   893→            const binaryStr = atob(rawBase64);
   894→            const bytes = new Uint8Array(binaryStr.length);
   895→            for (let i = 0; i < binaryStr.length; i++) {
   896→                bytes[i] = binaryStr.charCodeAt(i);
   897→            }
   898→            const blob = new Blob([bytes], { type: 'application/pdf' });
   899→            const blobUrl = URL.createObjectURL(blob);
   900→
   901→            // Use hidden iframe for reliable cross-browser printing
   902→            const iframe = document.createElement('iframe');
   903→            iframe.style.position = 'fixed';
   904→            iframe.style.right = '0';
   905→            iframe.style.bottom = '0';
   906→            iframe.style.width = '0';
   907→            iframe.style.height = '0';
   908→            iframe.style.border = 'none';
   909→            iframe.style.opacity = '0';
   910→            iframe.style.pointerEvents = 'none';
   911→            document.body.appendChild(iframe);
   912→
   913→            iframe.onload = function() {
   914→                setTimeout(() => {
   915→                    try {
   916→                        iframe.contentWindow.print();
   917→                    } catch(e) {
   918→                        console.warn('iframe print failed, trying window.open:', e);
   919→                        // Fallback to new window
   920→                        const printWindow = window.open(blobUrl, '_blank');
   921→                        if (printWindow) {
   922→                            printWindow.addEventListener('load', function() {
   923→                                setTimeout(() => { printWindow.print(); }, 800);
   924→                            });
   925→                        } else {
   926→                            app.downloadLastPdf();
   927→                        }
   928→                    }
   929→                    // Cleanup iframe after printing
   930→                    setTimeout(() => {
   931→                        document.body.removeChild(iframe);
   932→                        URL.revokeObjectURL(blobUrl);
   933→                    }, 5000);
   934→                }, 500);
   935→            };
   936→            iframe.src = blobUrl;
   937→        } catch(e) {
   938→            console.error('Print failed:', e);
   939→            app.downloadLastPdf();
   940→        }
   941→    },
