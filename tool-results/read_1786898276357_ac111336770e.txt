    69→async def _ensure_admin_user(app: FastAPI) -> None:
    70→    """Startup hook: handle first boot and migrate orphan threads otherwise.
    71→
    72→    After admin creation, migrate orphan threads from the LangGraph
    73→    store (metadata.user_id unset) to the admin account. This is the
    74→    "no-auth → with-auth" upgrade path: users who ran DeerFlow without
    75→    authentication have existing LangGraph thread data that needs an
    76→    owner assigned.
    77→        First boot (no admin exists):
    78→            - Does NOT create any user accounts automatically.
    79→            - The operator must visit ``/setup`` to create the first admin.
    80→
    81→    Subsequent boots (admin already exists):
    82→      - Runs the one-time "no-auth → with-auth" orphan thread migration for
    83→        existing LangGraph thread metadata that has no user_id.
    84→
    85→    No SQL persistence migration is needed: the four user_id columns
    86→    (threads_meta, runs, run_events, feedback) only come into existence
    87→    alongside the auth module via create_all, so freshly created tables
    88→    never contain NULL-owner rows.
    89→    """
    90→    from sqlalchemy import select
    91→
    92→    from app.gateway.deps import get_local_provider
    93→    from deerflow.persistence.engine import get_session_factory
    94→    from deerflow.persistence.user.model import UserRow
    95→
    96→    try:
    97→        provider = get_local_provider()
    98→    except RuntimeError:
    99→        # Auth persistence may not be initialized in some test/boot paths.
   100→        # Skip admin migration work rather than failing gateway startup.
   101→        logger.warning("Auth persistence not ready; skipping admin bootstrap check")
   102→        return
   103→
   104→    sf = get_session_factory()
   105→    if sf is None:
   106→        return
   107→
   108→    admin_count = await provider.count_admin_users()
   109→
   110→    if admin_count == 0:
   111→        logger.info("=" * 60)
   112→        logger.info("  First boot detected — no admin account exists.")
   113→        logger.info("  Visit /setup to complete admin account creation.")
   114→        logger.info("=" * 60)
   115→        return
   116→
   117→    # Admin already exists — run orphan thread migration for any
   118→    # LangGraph thread metadata that pre-dates the auth module.
   119→    async with sf() as session:
   120→        stmt = select(UserRow).where(UserRow.system_role == "admin").limit(1)
   121→        row = (await session.execute(stmt)).scalar_one_or_none()
   122→
   123→    if row is None:
   124→        return  # Should not happen (admin_count > 0 above), but be safe.
   125→
   126→    admin_id = str(row.id)
   127→
   128→    # LangGraph store orphan migration — non-fatal.
   129→    # This covers the "no-auth → with-auth" upgrade path for users
   130→    # whose existing LangGraph thread metadata has no user_id set.
   131→    store = getattr(app.state, "store", None)
   132→    if store is not None:
   133→        try:
   134→            migrated = await _migrate_orphaned_threads(store, admin_id)
   135→            if migrated:
   136→                logger.info("Migrated %d orphan LangGraph thread(s) to admin", migrated)
   137→        except Exception:
   138→            logger.exception("LangGraph thread migration failed (non-fatal)")

   141→async def _iter_store_items(store, namespace, *, page_size: int = 500):
   142→    """Paginated async iterator over a LangGraph store namespace.
   143→
   144→    Replaces the old hardcoded ``limit=1000`` call with a cursor-style
   145→    loop so that environments with more than one page of orphans do
   146→    not silently lose data. Terminates when a page is empty OR when a
   147→    short page arrives (indicating the last page).
   148→    """
   149→    offset = 0
   150→    while True:
   151→        batch = await store.asearch(namespace, limit=page_size, offset=offset)
   152→        if not batch:
   153→            return
   154→        for item in batch:
   155→            yield item
   156→        if len(batch) < page_size:
   157→            return
   158→        offset += page_size
   159→

   161→async def _migrate_orphaned_threads(store, admin_user_id: str) -> int:
   162→    """Migrate LangGraph store threads with no user_id to the given admin.
   163→
   164→    Uses cursor pagination so all orphans are migrated regardless of
   165→    count. Returns the number of rows migrated.
   166→    """
   167→    migrated = 0
   168→    async for item in _iter_store_items(store, ("threads",)):
   169→        metadata = item.value.get("metadata", {})
   170→        if not metadata.get("user_id"):
   171→            metadata["user_id"] = admin_user_id
   172→            item.value["metadata"] = metadata
   173→            await store.aput(("threads",), item.key, item.value)
   174→            migrated += 1
   175→    return migrated

   178→async def _warm_memory_retrieval(manager) -> None:
   179→    """Rebuild the derived retrieval index without delaying Gateway readiness."""
   180→    try:
   181→        rebuilt = await asyncio.to_thread(manager.warm_retrieval)
   182→        if rebuilt:
   183→            logger.info("Memory retrieval index rebuilt successfully")
   184→        else:
   185→            logger.warning("Memory retrieval index rebuild failed; scoped searches will retry lazily")
   186→    except Exception:
   187→        logger.warning("Memory retrieval index rebuild skipped", exc_info=True)

   193→

   201→    try:

   205→        logger.info("Configuration loaded successfully")
   206→        warn_if_auth_disabled_enabled()
   207→    except Exception as e:
   208→        error_msg = f"Failed to load configuration during gateway startup: {e}"
   209→        logger.exception(error_msg)
   210→        raise RuntimeError(error_msg) from e
   211→    config = get_gateway_config()
   212→    logger.info(f"Starting API Gateway on {config.host}:{config.port}")
   213→
   214→    from deerflow.skills.projection import ensure_public_skill_projection
   215→
   216→    public_projection_ready = await asyncio.to_thread(ensure_public_skill_projection, app_config=startup_config)
   217→    if public_projection_ready:
   218→        logger.info("Ensured the public skill projection; user projections repair lazily on sandbox acquire")
   219→
   220→    # Agent observability (Monocle). Off by default; enabled with
   221→    # MONOCLE_TRACING. Initialized here at startup — not at import time — so a
   222→    # plain `import deerflow.agents` never installs a process-global tracer.
   223→    # Unlike LangSmith/Langfuse, whose validation failures abort the agent run,
   224→    # a bad Monocle config only logs: the Gateway keeps serving without tracing.
   225→    try:
   226→        setup_monocle_tracing_if_enabled()
   227→    except Exception:  # observability must never break startup
   228→        logger.exception("Monocle tracing setup failed; continuing without it")
   229→
   230→    # Rebuild the derived memory retrieval index in the background. Scoped
   231→    # searches remain correct while this runs because DeerMem lazily rebuilds
   232→    # the requested scope when the full warm-up has not completed yet.
   233→    retrieval_warm_task: asyncio.Task[None] | None = None
   234→    try:
   235→        from deerflow.agents.memory import get_memory_manager
   236→
   237→        if startup_config.memory.enabled:
   238→            manager = await asyncio.to_thread(get_memory_manager)
   239→            warm_retrieval = getattr(manager, "warm_retrieval", None)
   240→            if callable(warm_retrieval):
   241→                retrieval_warm_task = asyncio.create_task(
   242→                    _warm_memory_retrieval(manager),
   243→                    name="memory-retrieval-warm-up",
   244→                )
   245→        else:
   246→            logger.info("Memory is disabled; skipping retrieval index rebuild")
   247→    except Exception:
   248→        logger.warning("Memory retrieval index rebuild skipped", exc_info=True)
   249→
   250→    # Pre-warm tiktoken encoding cache so the first memory-injection request
   251→    # never blocks on the BPE data download (which hits an OpenAI/Azure URL
   252→    # that may be unreachable in restricted networks — see issue #3402).
   253→    # Warm-up runs via the manager's `warm()` tier-3 hook. DeerMem.warm re-checks
   254→    # token_counting=="char" and returns early, so char-mode backends never touch
   255→    # tiktoken (avoids even the 5s probe in network-restricted deployments - see
   256→    # issue #3429). A backend with nothing to warm (e.g. noop) returns None from
   257→    # the base default -- log "skipping" instead of the misleading "warmed
   258→    # successfully" so the log reflects what actually happened.
   259→    try:
   260→        from deerflow.agents.memory import get_memory_manager
   261→
   262→        manager = await asyncio.to_thread(get_memory_manager)
   263→        warmed = await asyncio.wait_for(
   264→            asyncio.to_thread(manager.warm),
   265→            timeout=5,
   266→        )
   267→        if warmed is None:
   268→            logger.info("Memory backend %s has nothing to warm; skipping tiktoken warm-up", type(manager).__name__)
   269→        elif warmed:
   270→            logger.info("tiktoken encoding cache warmed successfully")
   271→        else:
   272→            logger.warning("tiktoken encoding cache warm-up failed; token counting will use character-based fallback until tiktoken loads successfully")
   273→    except TimeoutError:
   274→        logger.warning("tiktoken encoding cache warm-up timed out; token counting will use character-based fallback until tiktoken loads successfully")
   275→    except Exception:
   276→        logger.warning("tiktoken warm-up skipped", exc_info=True)
   277→
   278→    try:
   279→        removed_upload_staging_files = await asyncio.to_thread(cleanup_stale_upload_staging_files)
   280→        if removed_upload_staging_files:
   281→            logger.info("Removed %d stale upload staging file(s)", removed_upload_staging_files)
   282→    except Exception:
   283→        logger.warning("Upload staging file cleanup skipped", exc_info=True)
   284→

   295→            from app.channels.service import start_channel_service
   296→

   298→            # below) rather than resolving `app.state.stream_bridge` here
   299→            # directly: `stream_bridge` is a STARTUP_ONLY_FIELDS singleton set
   300→            # once, above, by `langgraph_runtime(app, startup_config)`, so
   301→            # either shape is safe by construction — the closure is just the
   302→            # more defensive/consistent-with-precedent form, and it is what
   303→            # ChannelManager's follow-up-drain watcher (issue #4121 Slice 2)
   304→            # uses to reach the same StreamBridge every other run consumer
   305→            # goes through `get_stream_bridge(request)` for.
   306→            channel_service = await start_channel_service(
   307→                startup_config,
   308→                get_stream_bridge=lambda: getattr(app.state, "stream_bridge", None),
   309→            )
   310→            logger.info("Channel service started: %s", channel_service.get_status())
   311→        except Exception:
   312→            logger.exception("No IM channels configured or channel service failed to start")
   313→
   314→        try:
   315→            from app.gateway.services import launch_scheduled_thread_run
   316→            from app.scheduler import ScheduledTaskService
   317→
   318→            if getattr(app.state, "scheduled_task_repo", None) is not None and getattr(app.state, "scheduled_task_run_repo", None) is not None:
   319→                scheduled_task_service = ScheduledTaskService(
   320→                    task_repo=app.state.scheduled_task_repo,
   321→                    task_run_repo=app.state.scheduled_task_run_repo,
   322→                    launch_run=lambda **kwargs: launch_scheduled_thread_run(app=app, **kwargs),
   323→                    poll_interval_seconds=startup_config.scheduler.poll_interval_seconds,
   324→                    lease_seconds=startup_config.scheduler.lease_seconds,
   325→                    max_concurrent_runs=startup_config.scheduler.max_concurrent_runs,
   326→                    multi_instance=startup_config.scheduler.multi_instance,
   327→                    run_lease_grace_seconds=startup_config.run_ownership.grace_seconds,
   328→                )
   329→                app.state.scheduled_task_service = scheduled_task_service
   330→                if startup_config.scheduler.enabled:
   331→                    await scheduled_task_service.start()
   332→        except Exception:
   333→            logger.exception("Failed to initialize scheduled task service")
   334→
   335→        from app.mcp_tasks import McpTaskService
   336→        from deerflow.config.extensions_config import ExtensionsConfig
   337→        from deerflow.config.mcp_tasks_config import McpTasksConfig
   338→        from deerflow.mcp.task_tool_caller import McpTaskToolCaller
   339→        from deerflow.mcp.tasks import (
   340→            ORDINARY_MCP_TASK_DRIVER,
   341→            McpTaskDriverRegistry,
   342→            OrdinaryMcpTaskDriver,
   343→        )
   344→        from deerflow.mcp.tasks.runtime import (
   345→            configured_task_toolset_count,
   346→            set_mcp_task_config_snapshot,
   347→            set_mcp_task_submitter,
   348→            validate_mcp_task_runtime_configuration,
   349→        )
   350→
   351→        task_extensions_config = ExtensionsConfig.from_file()
   352→        mcp_tasks_config = getattr(startup_config, "mcp_tasks", McpTasksConfig())
   353→        mcp_task_repo = getattr(app.state, "mcp_task_repo", None)
   354→        set_mcp_task_submitter(None)
   355→        set_mcp_task_config_snapshot(task_extensions_config)
   356→        validate_mcp_task_runtime_configuration(
   357→            mcp_tasks_config=mcp_tasks_config,
   358→            extensions_config=task_extensions_config,
   359→            repository_available=mcp_task_repo is not None,
   360→        )
   361→        if mcp_task_repo is not None:
   362→            mcp_task_drivers = McpTaskDriverRegistry()
   363→            if configured_task_toolset_count(task_extensions_config):
   364→                mcp_task_drivers.register(
   365→                    ORDINARY_MCP_TASK_DRIVER,
   366→                    OrdinaryMcpTaskDriver(McpTaskToolCaller(task_extensions_config)),
   367→                )
   368→            mcp_task_service = McpTaskService(
   369→                repository=mcp_task_repo,
   370→                drivers=mcp_task_drivers,
   371→                poll_interval_seconds=mcp_tasks_config.poll_interval_seconds,
   372→                lease_seconds=mcp_tasks_config.lease_seconds,
   373→                max_concurrent_polls=mcp_tasks_config.max_concurrent_polls,
   374→                max_poll_backoff_seconds=mcp_tasks_config.max_poll_backoff_seconds,
   375→                input_required_poll_interval_seconds=mcp_tasks_config.input_required_poll_interval_seconds,
   376→                tracking_degraded_after_errors=mcp_tasks_config.tracking_degraded_after_errors,
   377→                max_result_bytes=mcp_tasks_config.max_result_bytes,
   378→                result_preview_max_chars=mcp_tasks_config.result_preview_max_chars,
   379→            )
   380→            app.state.mcp_task_drivers = mcp_task_drivers
   381→            app.state.mcp_task_service = mcp_task_service
   382→            if mcp_tasks_config.enabled:
   383→                await mcp_task_service.start()
   384→                set_mcp_task_submitter(mcp_task_service)
   385→
   386→        yield
   387→
   388→        try:
   389→            await auth.close_oidc_service()
   390→        except Exception:
   391→            logger.exception("Failed to close OIDC service")
   392→
   393→        # Stop channel service on shutdown (bounded to prevent worker hang)
   394→        try:
   395→            from app.channels.service import stop_channel_service
   396→
   397→            await asyncio.wait_for(
   398→                stop_channel_service(),
   399→                timeout=_SHUTDOWN_HOOK_TIMEOUT_SECONDS,
   400→            )
   401→        except TimeoutError:
   402→            logger.warning(
   403→                "Channel service shutdown exceeded %.1fs; proceeding with worker exit.",
   404→                _SHUTDOWN_HOOK_TIMEOUT_SECONDS,
   405→            )
   406→        except Exception:
   407→            logger.exception("Failed to stop channel service")
   408→
   409→        if getattr(app.state, "scheduled_task_service", None) is not None:
   410→            try:
   411→                await app.state.scheduled_task_service.stop()
   412→            except Exception:
   413→                logger.exception("Failed to stop scheduled task service")
   414→
   415→        if getattr(app.state, "mcp_task_service", None) is not None:
   416→            try:
   417→                await app.state.mcp_task_service.stop()
   418→            except Exception:
   419→                logger.exception("Failed to stop MCP task service")
   420→            finally:
   421→                from deerflow.mcp.tasks.runtime import set_mcp_task_submitter
   422→
   423→                set_mcp_task_submitter(None)
   424→        from deerflow.mcp.tasks.runtime import set_mcp_task_config_snapshot
   425→
   426→        set_mcp_task_config_snapshot(None)
   427→
   428→        try:
   429→            from deerflow.community.browser_automation import get_browser_session_manager
   430→
   431→            closed = await asyncio.wait_for(
   432→                get_browser_session_manager().close_all_sessions(),
   433→                timeout=_SHUTDOWN_HOOK_TIMEOUT_SECONDS,
   434→            )
   435→            if closed:
   436→                logger.info("Closed %d browser session(s)", closed)
   437→        except TimeoutError:
   438→            logger.warning(
   439→                "Browser session shutdown exceeded %.1fs; proceeding with worker exit.",
   440→                _SHUTDOWN_HOOK_TIMEOUT_SECONDS,
   441→            )
   442→        except Exception:
   443→            logger.exception("Failed to close browser sessions")
   444→
   445→        # Drain the memory backend's pending-update buffer before the worker
   446→        # exits (best-effort, bounded). IM channels and the scheduler are
   447→        # already stopped above, so no new IM/scheduler updates arrive during
   448→        # the drain; the LangGraph runtime / in-flight HTTP requests can still
   449→        # complete memory enqueues in a narrow window, but anything added after
   450→        # the drain copies the buffer only resets the debounce Timer
   451→        # (best-effort, same as today).
   452→        #
   453→        # No host-level pending/processing guard: ``shutdown_flush``
   454→        # short-circuits on a truly idle buffer (returns True immediately), so
   455→        # calling it unconditionally is cheap and keeps the in-flight-worker
   456→        # race entirely inside the backend (where the buffer lives) -- the host
   457→        # cannot "forget" that case the way a ``pending_count > 0``-only guard
   458→        # would (review #6 on the original PR).
   459→        #
   460→        # K8s caveat: ``shutdown_flush_timeout_seconds`` must fit inside the
   461→        # pod's ``terminationGracePeriodSeconds`` (channel stop + browser
   462→        # session close + the brief retrieval-warm wait + this drain + buffer),
   463→        # set on the gateway Helm deployment -- or K8s SIGKILLs the drain
   464→        # mid-flight and the loss this is fixing is silently re-introduced.
   465→        # The retrieval index is derived from canonical memory files, so its
   466→        # wait is independently capped and never consumes the flush budget.
   467→        retrieval_warm_finished = True
   468→        if retrieval_warm_task is not None and not retrieval_warm_task.done():
   469→            try:
   470→                await asyncio.wait_for(
   471→                    asyncio.shield(retrieval_warm_task),
   472→                    timeout=min(
   473→                        _RETRIEVAL_WARM_SHUTDOWN_TIMEOUT_SECONDS,
   474→                        startup_config.memory.shutdown_flush_timeout_seconds,
   475→                    ),
   476→                )
   477→            except TimeoutError:
   478→                retrieval_warm_finished = False
   479→                logger.warning("Memory retrieval index rebuild is still running; leaving its connection open during shutdown")
   480→
   481→        manager = None
   482→        try:
   483→            # Memory shutdown runs on a worker thread and can trigger detached
   484→            # system-model callbacks. Stop accepting those callbacks before
   485→            # flushing, while keeping the registered loop alive for awaited
   486→            # task hooks until langgraph_runtime drains runs and subagents.
   487→            from deerflow.extensions.notify import suspend_extension_system_observations
   488→
   489→            suspend_extension_system_observations()
   490→        except Exception:
   491→            logger.debug("Failed to suspend extension system observations (non-fatal)", exc_info=True)
   492→
   493→        try:
   494→            app_cfg = get_app_config()
   495→            if app_cfg.memory.enabled:
   496→                from deerflow.agents.memory import get_memory_manager
   497→
   498→                manager = await asyncio.to_thread(get_memory_manager)
   499→                flush_timeout = app_cfg.memory.shutdown_flush_timeout_seconds
   500→                completed = await asyncio.to_thread(manager.shutdown_flush, flush_timeout)
   501→                if completed:
   502→                    logger.info("Memory queue flush completed within %.1fs", flush_timeout)
   503→                else:
   504→                    logger.warning(
   505→                        "Memory queue flush did not finish within %.1fs; remaining updates may be lost",
   506→                        flush_timeout,
   507→                    )
   508→        except Exception:
   509→            logger.exception("Failed to flush memory queue on shutdown")
   510→        finally:
   511→            close = getattr(manager, "close", None)
   512→            if callable(close) and retrieval_warm_finished:
   513→                try:
   514→                    await asyncio.to_thread(close)
   515→                except Exception:
   516→                    logger.exception("Failed to close memory backend on shutdown")
   517→
   518→    logger.info("Shutting down API Gateway")

   627→    # run id the Gateway returns in a non-safelisted response header; without
   628→    # exposing it the SDK never reports a created run, so a new thread keeps its

   797→def _resolve_trace_enabled_for_app_construction() -> bool:
