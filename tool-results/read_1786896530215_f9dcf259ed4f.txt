  2105→# Run Ownership Configuration
  2106→# ============================================================================
  2107→# Controls cross-process run ownership for multi-worker deployments.
  2108→# When GATEWAY_WORKERS > 1, each worker claims runs with a lease; the heartbeat
  2109→# renews leases, and reconciliation recovers orphaned runs from crashed workers.
  2110→#
  2111→# CLOCK-SYNC REQUIREMENT (multi-worker only): reconciliation compares another
  2112→# worker's UTC lease timestamp against this worker's datetime.now(UTC). Worker
  2113→# clocks MUST be synced (NTP / chrony / systemd-timesyncd — default on K8s and
  2114→# cloud VMs) within a few seconds. grace_seconds is the skew budget; worst case
  2115→# (owning worker's heartbeat just about to fire), a peer whose clock is more
  2116→# than grace_seconds ahead can mis-reclaim a still-live run as an orphan. Raise
  2117→# grace_seconds if your environment cannot keep clocks within a few seconds;
  2118→# the trade-off is longer recovery latency for genuinely dead workers
  2119→# (lease_seconds + grace_seconds from last heartbeat to reclaim).
  2120→#
  2121→# The owning worker itself does NOT use grace_seconds as extra execution time.
  2122→# If renewal cannot be confirmed before lease_seconds expires, it cancels the
  2123→# local run and suppresses durable finalization; grace_seconds only delays when
  2124→# a peer may reclaim the expired row.
  2125→
  2126→run_ownership:
  2127→  lease_seconds: 30           # Seconds before a run lease expires if not renewed.
  2128→                              # Heartbeat renews every lease_seconds / 3.

  2439→# ============================================================================
  2440→# LLM Call Concurrency Configuration
  2441→# ============================================================================
  2442→# Cap the number of concurrently in-flight LLM calls process-wide. A provider
  2443→# burst-rate (limit_burst_rate) error fires on the *slope* of the request rate,
  2444→# not on a static quota - so the morning peak (e.g. 08:30) ramping from ~0 to
  2445→# full throttle in seconds gets rejected even when total RPM is within budget.
  2446→# Capping concurrency caps that slope. Retries alone make it worse (they add
  2447→# demand inside the very burst being throttled); pair this cap with the
  2448→# decorrelated-jitter backoff already built into the LLM error-handling
  2449→# middleware, and ideally an nginx `limit_req` at the ingress.
  2450→#
  2451→# 0 disables the cap (default) - existing deployments see no behavior change.
  2452→#
  2453→# Per-process, not per-cluster: the cap bounds in-flight LLM calls within ONE
  2454→# gateway process. With GATEWAY_WORKERS > 1 the aggregate cap across the
  2455→# deployment is effectively `max_concurrent_calls * GATEWAY_WORKERS`, and a
  2456→# multi-node rollout multiplies it further - size the per-process value with
  2457→# that in mind (and pair it with an nginx `limit_req` at the ingress for a
  2458→# true cluster-wide slope cap).
  2459→#
  2460→# Startup-only: the cap is captured at the first LLM run and frozen for the
  2461→# process lifetime. Editing `max_concurrent_calls` here takes effect only after
  2462→# a gateway restart; the other `llm_call.*` knobs below remain hot-reloadable.
  2463→# Freezing the cap avoids the downscale and config-freshness races that a
  2464→# runtime-mutable, process-wide/cross-loop limiter would otherwise hit (a
  2465→# lowered cap could keep admitting queued waiters, or a stale config snapshot
  2466→# constructed after a fresher one could restore a higher cap).
  2467→
  2468→# llm_call:
  2469→#   # Max concurrently in-flight LLM calls across the whole process (default: 0 = disabled)
  2470→#   max_concurrent_calls: 0
  2471→#   # Max LLM call attempts for retriable transient errors, 1 = no retry (default: 3)
  2472→#   retry_max_attempts: 3
  2473→#   # Base delay (ms) for the decorrelated-jitter retry backoff (default: 1000)
  2474→#   retry_base_delay_ms: 1000
  2475→#   # Hard cap (ms) on any single retry backoff delay (default: 8000)
  2476→#   retry_cap_delay_ms: 8000
  2477→#   # Backoff base (ms) used ONLY for burst-rate (limit_burst_rate) 429s - higher
  2478→#   # than retry_base_delay_ms so the single burst retry lands after the throttle
  2479→#   # window subsides. Ignored when the provider sends Retry-After (default: 5000)
  2480→#   burst_retry_base_delay_ms: 5000

  2550→# Extension packages, loaded once at startup in the order listed here. Each
  2551→# entry names an install entry point as "module.path:install"; the `config`
  2552→# block is handed to that package verbatim and validated by the package itself.
  2553→#
  2554→# Prefer the extension manager over editing this list by hand:
  2555→#   make extension-install SOURCE="deerflow-extension-acme==1.2.3"
  2556→#   make extension-install SOURCE="git+https://github.com/acme/deerflow-extension-acme.git@<commit>"
  2557→#   make extension-install SOURCE="/absolute/path/to/local-extension"
  2558→#   make extension-list
  2559→#   make extension-enable NAME=acme
  2560→#   make extension-disable NAME=acme
  2561→#   make extension-remove NAME=acme
  2562→# The installer updates backend/pyproject.toml, backend/uv.lock, and this block
  2563→# together. It also asks the operator to confirm that the source is trusted;
  2564→# extensions and their build systems execute with Gateway privileges.
  2565→#
  2566→# Every install, enable, disable, remove, or manual edit requires a Gateway
  2567→# restart. Development launchers may fetch missing locked artifacts before
  2568→# handoff; a built production Gateway never installs plugins during startup.
  2569→# Local directories are copied to backend/extensions/sources as deployable
  2570→# snapshots; use a pinned version or public HTTPS Git commit for reproducible
  2571→# remote sources. SSH Git URLs are rejected because the stock Docker builder
  2572→# does not forward host SSH credentials.
  2573→# Loading remains explicit: a package installed outside the manager does nothing
  2574→# until it is listed here, and list order fixes contribution ordering.
  2575→#
  2576→# This is deliberately separate from the `extensions:` block above (MCP servers,
  2577→# skills, config-declared middlewares). That one is backed by
  2578→# extensions_config.json, which the Gateway can rewrite through an HTTP
  2579→# endpoint; a list that causes code to be imported must stay in this
  2580→# operator-controlled file only.
  2581→#
  2582→# Plugins can contribute semantically placed middleware, task-lifecycle hooks,
  2583→# system-model observers, Gateway-lifetime services, and eager FastAPI HTTP routers.
  2584→# See examples/deerflow-extension-example for a package that exercises all five.
  2585→#
  2586→# plugins:
  2587→#   - name: example                       # PEP 621 entry-point name
