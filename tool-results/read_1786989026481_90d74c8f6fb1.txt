   923→@router.post("/{thread_id}/runs/{run_id}/cancel")
   924→@require_permission("runs", "cancel", owner_check=True, require_existing=True)
   925→async def cancel_run(
   926→    thread_id: ThreadId,
   927→    run_id: str,
   928→    request: Request,
   929→    wait: bool = Query(default=False, description="Block until run completes after cancel"),
   930→    action: Literal["interrupt", "rollback"] = Query(default="interrupt", description="Cancel action"),
   931→) -> Response:
   932→    """Cancel a running or pending run.
   933→
   934→    - action=interrupt: Stop execution, keep current checkpoint (can be resumed)
   935→    - action=rollback: Stop execution, revert to pre-run checkpoint state
   936→    - wait=true: Block until the run fully stops, return 204
   937→    - wait=false: Return immediately with 202
   938→
   939→    In multi-worker deployments, a cancel landing on a non-owning worker
   940→    durably notifies the owner when its lease is live, or takes over and
   941→    terminalizes the run when that lease has expired.
   942→    """
   943→    run_mgr = get_run_manager(request)
   944→    record = await run_mgr.get(run_id)
   945→    if record is None or record.thread_id != thread_id:
   946→        raise HTTPException(status_code=404, detail=f"Run {run_id} not found")
   947→
   948→    outcome = await run_mgr.cancel(run_id, action=action)
   949→
   950→    # Success paths — the run was cancelled locally, durably requested from
   951→    # a live owner, or taken over from a dead worker.
   952→    if outcome in (
   953→        CancelOutcome.cancelled,
   954→        CancelOutcome.requested,
   955→        CancelOutcome.taken_over,
   956→    ):
   957→        if wait and record.task is not None:
   958→            try:
   959→                await record.task
   960→            except asyncio.CancelledError:
   961→                pass
   962→            return Response(status_code=204)
   963→        if wait and outcome == CancelOutcome.requested:
   964→            bridge = get_stream_bridge(request)
   965→            if record.store_only and bridge.supports_cross_process:
   966→                completed = await wait_for_run_completion(
   967→                    bridge,
   968→                    record,
   969→                    request,
   970→                    run_mgr,
   971→                )
   972→                if completed:
   973→                    return Response(status_code=204)
   974→        return Response(status_code=202)
   975→
   976→    if outcome == CancelOutcome.lease_valid_elsewhere:
   977→        await _raise_lease_valid_elsewhere(run_id, run_mgr, record)
   978→
   979→    # not_cancellable, not_active_locally, unknown
   980→    raise HTTPException(status_code=409, detail=_cancel_conflict_detail(run_id, record))

  1006→# Register GET and POST as separate routes so each method gets a unique OpenAPI
  1007→# operationId. ``api_route(methods=["GET", "POST"])`` shares one route registration
  1008→# across both methods, which makes FastAPI emit the same ``operationId`` twice and
  1009→# warn about a duplicate operation id during OpenAPI generation.
  1010→@router.get("/{thread_id}/runs/{run_id}/stream", response_model=None)
  1011→@router.post("/{thread_id}/runs/{run_id}/stream", response_model=None)
  1012→@require_permission("runs", "read", owner_check=True)
  1013→async def stream_existing_run(
  1014→    thread_id: ThreadId,
  1015→    run_id: str,
  1016→    request: Request,
  1017→    action: Literal["interrupt", "rollback"] | None = Query(default=None, description="Cancel action"),
  1018→    wait: int = Query(default=0, description="Block until cancelled (1) or return immediately (0)"),
  1019→):
  1020→    """Join an existing run's SSE stream (GET), or cancel-then-stream (POST).
  1021→
  1022→    The LangGraph SDK's ``joinStream`` and ``useStream`` stop button both use
  1023→    ``POST`` to this endpoint.  When ``action=interrupt`` or ``action=rollback``
  1024→    is present the run is cancelled first; the response then streams any
  1025→    remaining buffered events so the client observes a clean shutdown.
  1026→    """
  1027→    run_mgr = get_run_manager(request)
  1028→    record = await run_mgr.get(run_id)
  1029→    if record is None or record.thread_id != thread_id:
  1030→        raise HTTPException(status_code=404, detail=f"Run {run_id} not found")
  1031→    bridge = get_stream_bridge(request)
  1032→    if record.store_only and action is None and not bridge.supports_cross_process:
  1033→        raise HTTPException(status_code=409, detail=f"Run {run_id} is not active on this worker and cannot be streamed")
  1034→
  1035→    # Cancel if an action was requested (stop-button / interrupt flow)
  1036→    if action is not None:
  1037→        outcome = await run_mgr.cancel(run_id, action=action)
  1038→        if outcome == CancelOutcome.taken_over:
  1039→            # The run was on another worker and is now marked ``error`` in the
  1040→            # store.  There is no local stream to drain — return immediately so
  1041→            # the client doesn't hang on an SSE subscription this worker can
  1042→            # never serve.
  1043→            return Response(status_code=202)
  1044→        if outcome not in (CancelOutcome.cancelled, CancelOutcome.requested):
  1045→            if outcome == CancelOutcome.lease_valid_elsewhere:
  1046→                await _raise_lease_valid_elsewhere(run_id, run_mgr, record)
  1047→            raise HTTPException(status_code=409, detail=_cancel_conflict_detail(run_id, record))
  1048→        if outcome == CancelOutcome.requested and record.store_only and not bridge.supports_cross_process:
  1049→            # The request is durable, but this bridge cannot observe the
  1050→            # owner's stream. Returning 202 is safer than hanging forever on
  1051→            # a process-local subscription.
  1052→            return Response(status_code=202)
  1053→        if wait and record.task is not None:
  1054→            try:
  1055→                await record.task
  1056→            except (asyncio.CancelledError, Exception):
  1057→                pass
  1058→            return Response(status_code=204)
  1059→        if wait and outcome == CancelOutcome.requested:
  1060→            completed = await wait_for_run_completion(
  1061→                bridge,
  1062→                record,
  1063→                request,
  1064→                run_mgr,
  1065→            )
  1066→            return Response(status_code=204 if completed else 202)
  1067→
  1068→    return StreamingResponse(
  1069→        sse_consumer(bridge, record, request, run_mgr),
  1070→        media_type="text/event-stream",
  1071→        headers={
  1072→            "Cache-Control": "no-cache",
  1073→            "Connection": "keep-alive",
  1074→            "X-Accel-Buffering": "no",
  1075→        },
  1076→    )

  1085→@require_permission("runs", "read", owner_check=True)
  1086→async def list_thread_messages(
  1087→    thread_id: ThreadId,
  1088→    request: Request,
  1089→    limit: int = Query(default=50, ge=1, le=200),
  1090→    before_seq: int | None = Query(default=None, ge=1),
  1091→    after_seq: int | None = Query(default=None, ge=1),
  1092→) -> list[dict]:
  1093→    """Return displayable messages for a thread (across all runs), with feedback attached."""
  1094→    # Resolve the caller once; it is needed both to scope the feedback query
  1095→    # below and to list the thread's runs for turn-duration injection.
  1096→    user_id = await get_current_user(request)
  1097→    run_mgr = get_run_manager(request)
  1098→    hidden_run_ids = await _default_history_hidden_run_ids(run_mgr, thread_id, user_id=user_id)
  1099→    messages, _ = await _scan_visible_thread_messages(
  1100→        thread_id,
  1101→        limit=limit,
  1102→        before_seq=before_seq,
  1103→        after_seq=after_seq,
  1104→        request=request,
  1105→        user_id=user_id,
  1106→        hidden_run_ids=hidden_run_ids,
  1107→        include_middleware=True,
  1108→        include_extra=False,
  1109→        batch_size=THREAD_MESSAGE_LEGACY_SCAN_BATCH,
  1110→    )
  1111→
  1112→    # Find the last AI message per run_id. AI messages are persisted by
  1113→    # RunJournal with event_type "llm.ai.response" (see runtime/journal.py);
  1114→    # the event store returns that value verbatim, so match on it here.
  1115→    last_ai_per_run: dict[str, int] = {}  # run_id -> index in messages list
  1116→    for i, msg in enumerate(messages):
  1117→        if msg.get("event_type") == "llm.ai.response":
  1118→            last_ai_per_run[msg["run_id"]] = i
  1119→
  1120→    # Attach feedback to the last AI message of each run. Only query when there
  1121→    # is an AI message to attach it to — threads with no completed AI turn yet
  1122→    # would otherwise pay for a grouped feedback lookup whose result is unused.
  1123→    feedback_map: dict[str, dict] = {}
  1124→    if last_ai_per_run:
  1125→        feedback_repo = get_feedback_repo(request)
  1126→        feedback_map = await feedback_repo.list_by_thread_grouped(thread_id, user_id=user_id)
  1127→
  1128→    last_ai_indices = set(last_ai_per_run.values())
  1129→    for i, msg in enumerate(messages):
  1130→        if i in last_ai_indices:
  1131→            run_id = msg["run_id"]
  1132→            fb = feedback_map.get(run_id)
  1133→            msg["feedback"] = (
  1134→                {
  1135→                    "feedback_id": fb["feedback_id"],
  1136→                    "rating": fb["rating"],
  1137→                    "comment": fb.get("comment"),
  1138→                }
  1139→                if fb
  1140→                else None
  1141→            )
  1142→        else:
  1143→            msg["feedback"] = None
  1144→
  1145→    runs = await run_mgr.list_by_thread(thread_id, user_id=user_id)
  1146→    run_durations = compute_run_durations(runs)
  1147→
  1148→    if run_durations:
  1149→        stamp_turn_duration_on_last_ai(messages, run_durations)
  1150→
  1151→    return messages

  1154→async def _scan_visible_thread_messages(
  1155→    thread_id: str,
  1156→    *,
  1157→    limit: int,
  1158→    before_seq: int | None,
  1159→    after_seq: int | None,
  1160→    request: Request,
  1161→    user_id: str | None,
  1162→    hidden_run_ids: set[str],
  1163→    include_middleware: bool,
  1164→    include_extra: bool,
  1165→    batch_size: int,
  1166→) -> tuple[list[dict[str, Any]], bool]:
  1167→    """Scan raw message rows until ``limit`` visible rows survive filtering."""
  1168→    event_store = get_run_event_store(request)
  1169→    needed = limit + 1 if include_extra else limit
  1170→
  1171→    if after_seq is not None:
  1172→        visible: list[dict[str, Any]] = []
  1173→        scan_after = after_seq
  1174→        while len(visible) < needed:
  1175→            raw = await event_store.list_messages(
  1176→                thread_id,
  1177→                limit=batch_size,
  1178→                after_seq=scan_after,
  1179→                user_id=user_id,
  1180→            )
  1181→            if not raw:
  1182→                break
  1183→            _validate_message_scan_rows(raw, thread_id=thread_id, scan_before=None, scan_after=scan_after)
  1184→            reached_before_bound = False
  1185→            for row in raw:
  1186→                if before_seq is not None and row["seq"] >= before_seq:
  1187→                    reached_before_bound = True
  1188→                    break
  1189→                if (not include_middleware and _is_thread_history_hidden_message_row(row)) or row.get("run_id") in hidden_run_ids:
  1190→                    continue
  1191→                visible.append(row)
  1192→                if len(visible) == needed:
  1193→                    break
  1194→            next_scan_after = max(row["seq"] for row in raw)
  1195→            if next_scan_after <= scan_after:
  1196→                _raise_non_advancing_message_scan(thread_id=thread_id, scan_before=None, scan_after=scan_after, next_cursor=next_scan_after, row_count=len(raw))
  1197→            scan_after = next_scan_after
  1198→            if reached_before_bound or len(raw) < batch_size:
  1199→                break
  1200→        has_more = len(visible) > limit
  1201→        return visible[:limit], has_more
  1202→
  1203→    visible_desc: list[dict[str, Any]] = []
  1204→    scan_before = before_seq
  1205→    while len(visible_desc) < needed:
  1206→        raw = await event_store.list_messages(
  1207→            thread_id,
  1208→            limit=batch_size,
  1209→            before_seq=scan_before,
  1210→            user_id=user_id,
  1211→        )
  1212→        if not raw:
  1213→            break
  1214→        _validate_message_scan_rows(raw, thread_id=thread_id, scan_before=scan_before, scan_after=None)
  1215→        for row in reversed(raw):
  1216→            if (not include_middleware and _is_thread_history_hidden_message_row(row)) or row.get("run_id") in hidden_run_ids:
  1217→                continue
  1218→            visible_desc.append(row)
  1219→            if len(visible_desc) == needed:
  1220→                break
  1221→        next_scan_before = min(row["seq"] for row in raw)
  1222→        if scan_before is not None and next_scan_before >= scan_before:
  1223→            _raise_non_advancing_message_scan(thread_id=thread_id, scan_before=scan_before, scan_after=None, next_cursor=next_scan_before, row_count=len(raw))
  1224→        scan_before = next_scan_before
  1225→        if len(raw) < batch_size:
  1226→            break
  1227→    has_more = len(visible_desc) > limit
  1228→    return list(reversed(visible_desc[:limit])), has_more

  1231→def _validate_message_scan_rows(

  1242→            thread_id,

  1244→            scan_after,
  1245→            len(rows),

  1295→async def _enrich_thread_message_page(
  1296→    thread_id: str,
  1297→    rows: list[dict[str, Any]],
  1298→    *,
  1299→    request: Request,
  1300→    user_id: str | None,
  1301→) -> list[dict[str, Any]]:
  1302→    """Attach run-scoped duration and feedback without mutating store rows."""
  1303→    data = deepcopy(rows)
  1304→    if not data:
  1305→        return data
  1306→
  1307→    run_ids = {row["run_id"] for row in data if isinstance(row.get("run_id"), str)}
  1308→    run_mgr = get_run_manager(request)
  1309→    records = await run_mgr.get_many_by_thread(thread_id, run_ids, user_id=user_id)
  1310→    run_durations = compute_run_durations(records.values())
  1311→
  1312→    event_store = get_run_event_store(request)
  1313→    last_ai_seq_by_run = await event_store.get_last_visible_ai_seq_by_run(thread_id, run_ids, user_id=user_id)
  1314→    feedback_map: dict[str, dict] = {}
  1315→    feedback_run_ids = {run_id for row in data if isinstance((run_id := row.get("run_id")), str) and row.get("seq") == last_ai_seq_by_run.get(run_id)}
  1316→    if feedback_run_ids:
  1317→        feedback_repo = get_feedback_repo(request)
  1318→        feedback_map = await feedback_repo.list_by_run_ids(thread_id, feedback_run_ids, user_id=user_id)
  1319→

  1332→    # ``turn_duration`` is the run's wall-clock lifetime, not model thinking
