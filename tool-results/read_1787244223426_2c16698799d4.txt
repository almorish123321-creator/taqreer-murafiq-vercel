    34→// Helper to compute remaining subscription days
    35→const getDaysRemaining = (expiresAt) => {
    36→    if (!expiresAt) return 0;
    37→    const now = new Date();
    38→    const expires = new Date(expiresAt);
    39→    const diffMs = expires - now;
    40→    const diffDays = Math.ceil(diffMs / (1000 * 60 * 60 * 24));
    41→    return diffDays > 0 ? diffDays : 0;
    42→};
    43→
    44→// Normalize subscriber object
    45→const normalizeSubscription = (user) => {
    46→    if (!user) return null;
    47→    const now = new Date();
    48→    
    49→    // Migration helper: if they have subscriptionDays > 0 but no expires date
    50→    if (user.subscriptionDays > 0 && !user.subscriptionExpires) {
    51→        const expires = new Date(now.getTime() + user.subscriptionDays * 24 * 60 * 60 * 1000);
    52→        user.subscriptionExpires = expires.toISOString();
    53→    }
    54→    
    55→    user.subscriptionDays = getDaysRemaining(user.subscriptionExpires);
    56→    return user;
    57→};

    78→// Find user subscription by Chat ID or Telegram Username
    79→const findSubscription = async (chatId, username, referrerId = null) => {
    80→    const data = await loadLocalSubscriptions();
    81→    const chatIdStr = chatId.toString();
    82→    const cleanedUsername = username ? username.replace(/^@/, '').toLowerCase() : null;
    83→    
    84→    let userSub = null;
    85→    let foundChatId = chatIdStr;
    86→    
    87→    // 1. Search by Username
    88→    if (cleanedUsername) {
    89→        for (const [cid, sub] of Object.entries(data.subscriptions)) {
    90→            if (sub.username && sub.username.toLowerCase() === cleanedUsername) {
    91→                userSub = sub;
    92→                foundChatId = cid;
    93→                break;
    94→            }
    95→        }
    96→    }
    97→    
    98→    // 2. Search by Chat ID
    99→    if (!userSub && data.subscriptions[chatIdStr]) {
   100→        userSub = data.subscriptions[chatIdStr];
   101→    }
   102→    
   103→    // 3. Normalize subscription or create new
   104→    if (userSub) {
   105→        userSub = normalizeSubscription(userSub);
   106→        if (cleanedUsername && userSub.username !== cleanedUsername) {
   107→            userSub.username = cleanedUsername;
   108→        }
   109→        
   110→        // If we matched a pending Username subscription, migrate it to the active Chat ID
   111→        if (foundChatId !== chatIdStr) {
   112→            delete data.subscriptions[foundChatId];
   113→            data.subscriptions[chatIdStr] = userSub;
   114→        }
   115→        
   116→        data.subscriptions[chatIdStr].updatedAt = new Date().toISOString();
   117→        await saveLocalSubscriptions(data);
   118→    } else {
   119→        const now = new Date();
   120→        const expires = new Date(now.getTime() + 365 * 24 * 60 * 60 * 1000);
   121→        userSub = {
   122→            points: 0,
   123→            subscriptionDays: 365,
   124→            subscriptionExpires: expires.toISOString(),
   125→            username: cleanedUsername,
   126→            reports: [],
   127→            referredBy: referrerId ? referrerId.toString() : null,
   128→            referralsCount: 0,
   129→            referralPoints: 0,
   130→            updatedAt: now.toISOString()
   131→        };
   132→        
   133→        // If referred by someone, increment their referralsCount
   134→        if (referrerId) {
   135→            const rId = referrerId.toString();
   136→            if (data.subscriptions[rId]) {
   137→                data.subscriptions[rId].referralsCount = (data.subscriptions[rId].referralsCount || 0) + 1;
   138→                data.subscriptions[rId].updatedAt = now.toISOString();
   139→            }
   140→        }
   141→        
   142→        data.subscriptions[chatIdStr] = userSub;
   143→        await saveLocalSubscriptions(data);
   144→    }
   145→    
   146→    return { chatId: chatIdStr, ...userSub };
   147→};

   149→// Add or renew subscription for Username
   150→const addSubscriptionByUsername = async (username, days) => {
   151→    const data = await loadLocalSubscriptions();
   152→    const cleaned = username.replace(/^@/, '').toLowerCase();
   153→    
   154→    let foundChatId = null;
   155→    let userSub = null;
   156→    
   157→    for (const [cid, sub] of Object.entries(data.subscriptions)) {
   158→        if (sub.username && sub.username.toLowerCase() === cleaned) {
   159→            userSub = sub;
   160→            foundChatId = cid;
   161→            break;
   162→        }
   163→    }
   164→    
   165→    const now = new Date();
   166→    let baseDate = now;
   167→    
   168→    if (userSub) {
   169→        userSub = normalizeSubscription(userSub);
   170→        if (userSub.subscriptionExpires) {
   171→            const currentExpires = new Date(userSub.subscriptionExpires);
   172→            if (currentExpires > now) {
   173→                baseDate = currentExpires;
   174→            }
   175→        }
   176→    } else {
   177→        userSub = {
   178→            points: 0,
   179→            subscriptionDays: 0,
   180→            subscriptionExpires: null,
   181→            username: cleaned,
   182→            reports: [],
   183→            referredBy: null,
   184→            referralsCount: 0,
   185→            referralPoints: 0,
   186→            updatedAt: now.toISOString()
   187→        };
   188→        foundChatId = `pending_${cleaned}`;
   189→    }
   190→    
   191→    const expires = new Date(baseDate.getTime() + days * 24 * 60 * 60 * 1000);
   192→    userSub.subscriptionExpires = expires.toISOString();
   193→    userSub.subscriptionDays = getDaysRemaining(userSub.subscriptionExpires);
   194→    userSub.updatedAt = now.toISOString();
   195→    
   196→    // Referral rewards!
   197→    if (userSub.referredBy && !userSub.referralAwarded) {
   198→        const referrerId = userSub.referredBy.toString();
   199→        if (data.subscriptions[referrerId]) {
   200→            // Determine reward points based on subscription days
   201→            let rewardPoints = 0;
   202→            if (days === 30) rewardPoints = 50;
   203→            else if (days === 90) rewardPoints = 150;
   204→            else if (days === 180) rewardPoints = 300;
   205→            else if (days >= 365) rewardPoints = 600;
   206→            
   207→            if (rewardPoints > 0) {
   208→                data.subscriptions[referrerId].referralPoints = (data.subscriptions[referrerId].referralPoints || 0) + rewardPoints;
   209→                data.subscriptions[referrerId].points = (data.subscriptions[referrerId].points || 0) + rewardPoints;
   210→                data.subscriptions[referrerId].updatedAt = now.toISOString();
   211→                userSub.referralAwarded = true; // prevent multiple awards from the same user's first activation
   212→                
   213→                // Notify referrer
   214→                try {
   215→                    await bot.sendMessage(referrerId, `🎁 لقد حصلت على ${rewardPoints} نقطة مجانية كمكافأة لأن المستخدم @${username} الذي قمت بدعوته قام بالاشتراك!`);
   216→                } catch (e) {
   217→                    console.warn('Could not notify referrer:', e.message);
   218→                }
   219→            }
   220→        }
   221→    }
   222→    
   223→    data.subscriptions[foundChatId] = userSub;
   224→    await saveLocalSubscriptions(data);
   225→    
   226→    return { chatId: foundChatId, ...userSub };
   227→};
