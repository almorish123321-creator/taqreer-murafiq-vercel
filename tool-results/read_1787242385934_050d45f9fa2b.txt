    28→// Helper to compute remaining subscription days
    29→const getDaysRemaining = (expiresAt) => {
    30→    if (!expiresAt) return 0;
    31→    const now = new Date();
    32→    const expires = new Date(expiresAt);
    33→    const diffMs = expires - now;
    34→    const diffDays = Math.ceil(diffMs / (1000 * 60 * 60 * 24));
    35→    return diffDays > 0 ? diffDays : 0;
    36→};

    38→// Normalize subscriber object
    39→const normalizeSubscription = (user) => {
    40→    if (!user) return null;
    41→    const now = new Date();
    42→    
    43→    // Migration helper: if they have subscriptionDays > 0 but no expires date
    44→    if (user.subscriptionDays > 0 && !user.subscriptionExpires) {
    45→        const expires = new Date(now.getTime() + user.subscriptionDays * 24 * 60 * 60 * 1000);
    46→        user.subscriptionExpires = expires.toISOString();
    47→    }
    48→    
    49→    user.subscriptionDays = getDaysRemaining(user.subscriptionExpires);
