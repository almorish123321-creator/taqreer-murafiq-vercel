   411→    formatGregorian(dateString) {
   412→        if(!dateString) return "";
   413→        const parts = dateString.split('-');
   414→        if(parts.length===3) return `${parts[2]}-${parts[1]}-${parts[0]}`;
   415→        return dateString;

   418→    formatAMPM(timeStr) {
   419→        if(!timeStr) return "";
   420→        let [hours, minutes] = timeStr.split(':');
   421→        hours = parseInt(hours);
   422→        let ampm = hours >= 12 ? 'PM' : 'AM';
   423→        hours = hours % 12;
   424→        hours = hours ? hours : 12; 
   425→        return `${hours}:${minutes} ${ampm}`;
   426→    },

   885→    printLastPdf() {
   886→        if (!app.state.lastPdfBase64) {
   887→            if(this.tg) this.tg.showAlert('لا يوجد تقرير للطباعة');
   888→            else alert('لا يوجد تقرير للطباعة');
   889→            return;
   890→        }
   891→        try {
   892→            // Extract raw base64 data (remove data:application/pdf;base64, prefix)
   893→            const rawBase64 = app.state.lastPdfBase64.split(',')[1] || app.state.lastPdfBase64;
   894→            
   895→            // Convert base64 to binary blob
   896→            const binaryStr = atob(rawBase64);
   897→            const bytes = new Uint8Array(binaryStr.length);
   898→            for (let i = 0; i < binaryStr.length; i++) {
   899→                bytes[i] = binaryStr.charCodeAt(i);
   900→            }
   901→            const blob = new Blob([bytes], { type: 'application/pdf' });
   902→            const blobUrl = URL.createObjectURL(blob);
   903→            
   904→            // Try opening in new window for printing (works outside Telegram)
   905→            const printWindow = window.open(blobUrl, '_blank');
   906→            if (printWindow) {
   907→                printWindow.addEventListener('load', function() {
   908→                    setTimeout(() => {
   909→                        try {
   910→                            printWindow.print();
   911→                        } catch(e) {
   912→                            console.warn('Print from new window failed:', e);
   913→                        }
   914→                    }, 800);
   915→                });
   916→                // Clean up blob URL after 30 seconds
   917→                setTimeout(() => URL.revokeObjectURL(blobUrl), 30000);
   918→            } else {
   919→                // window.open blocked (Telegram WebView) — fallback to download
   920→                console.warn('window.open blocked, falling back to download');
   921→                app.downloadLastPdf();
   922→            }
   923→        } catch(e) {
   924→            console.error('Print failed:', e);
   925→            // Final fallback: download the file
   926→            app.downloadLastPdf();
   927→        }
   928→    },
