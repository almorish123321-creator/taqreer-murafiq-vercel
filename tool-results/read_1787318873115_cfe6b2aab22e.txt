    30→    30→// ============================================================
    31→    31→// Arabic BiDi + Reshaping — exact port of the Python bot's
    32→    32→// `safe_arabic_mixed` (line 59 of pdf_generator_updated (2).py).
    33→    33→//
    34→    34→// The bot uses `arabic_reshaper.reshape(text)` + `bidi.algorithm.get_display(text)`
    35→    35→// to convert logical-order Arabic into visual-order (post-BiDi) display text.
    36→    36→// We replicate that here with the JS equivalents: `arabic-reshaper` (Louay
    37→    37→// Alakkad's port) and `bidi-js` (lojjic's UBA implementation).
    38→    38→//
    39→    39→// The output string is in VISUAL LTR order: read left-to-right on screen,
    40→    40→// but each Arabic run is already reversed and shape-substituted to its
    41→    41→// presentation form (U+FE70-U+FEFF range). This means the renderer must
    42→    42→// NOT apply PDFKit's `features: ["rtla"]` (which would re-shape already-
    43→    43→// shaped chars and break them).
    44→    44→// ============================================================
    45→    45→const bidiEngine = bidiFactory();
    46→    46→function processArabicBiDi(text: string): string {
    47→    47→  if (!text) return "";
    48→    48→  try {
    49→    49→    const reshaped = arabicReshaper.convertArabic(text);
    50→    50→    const levels = bidiEngine.getEmbeddingLevels(reshaped);
    51→    51→    return bidiEngine.getReorderedString(reshaped, levels);
    52→    52→  } catch (e) {
    53→    53→    return text;
    54→    54→  }
    55→    55→}

   259→   259→
   260→   260→  const drawTextAr = (
   261→   261→    text: string,
   262→   262→    x: number,
   263→   263→    y: number,
   264→   264→    options: DrawOpts = {},
   265→   265→  ) => {
   266→   266→    const fontToUse = options.weight === "bold" ? fontArBold : fontArReg;
   267→   267→    const fontEnUse = options.weight === "bold" ? fontEnBold : fontEnReg;
   268→   268→
   269→   269→    // ============================================================
   270→   270→    // SPECIAL CASE: text contains "/" (forward slash).
   271→   271→    // Noto Sans Arabic does NOT have the U+002F glyph — it renders as
   272→   272→    // tofu (empty box). We split the text on "/" and render each Arabic
   273→   273→    // piece with the Arabic font, the slash itself (with surrounding
   274→   274→    // extra spaces) with Times-Bold.
   275→   275→    //
   276→   276→    // Visual layout: "رقم الهوية  /  الإقامة" — EXTRA space BEFORE and
   277→   277→    // AFTER the slash (two spaces each side, per user request), and the
   278→   278→    // slash is BOLD (Times-Bold) to stand out from the regular Arabic.
   279→   279→    //
   280→   280→    // Vertical alignment: Arabic fonts (Noto Sans Arabic) sit lower on
   281→   281→    // the baseline than Times-Roman at the same font size, so the slash
   282→   282→    // would appear higher than the Arabic text. We measure the actual
   283→   283→    // baseline of each font and offset the slash Y so its baseline
   284→   284→    // matches the Arabic baseline.
   285→   285→    // ============================================================
   286→   286→    if (useArabicFont && String(text).includes("/")) {
   287→   287→      const fontSize = options.fontSize || 12;
   288→   288→      const color = options.color || "#000000";
   289→   289→      const pieces = String(text).split("/");
   290→   290→      const trimmedPieces = pieces.map((p) => p.trim());
   291→   291→
   292→   292→      // The visual line: piece0 + "  /  " + piece1 — TWO spaces around
   293→   293→      // the slash for extra breathing room, and the slash is always
   294→   294→      // Times-Bold (regardless of options.weight) per user request.
   295→   295→      const slashWithSpaces = "  /  ";
   296→   296→      const numSlashes = pieces.length - 1;
   297→   297→      const slashFont = fontEnBold; // always bold for the slash
   298→   298→
   299→   299→      // Measure each piece with its own font
   300→   300→      doc.font(fontToUse).fontSize(fontSize);
   301→   301→      const arabicWidths = trimmedPieces.map((p) => doc.widthOfString(p));
   302→   302→      const arabicH = doc.heightOfString("م");
   303→   303→
   304→   304→      doc.font(slashFont).fontSize(fontSize);
   305→   305→      const slashGroupWidth = doc.widthOfString(slashWithSpaces);
   306→   306→      const slashH = doc.heightOfString("/");
   307→   307→
   308→   308→      // Vertical offset: shift the slash group DOWN to align with Arabic baseline
   309→   309→      const yOffset = arabicH - slashH;
   310→   310→
   311→   311→      const totalWidth =
   312→   312→        arabicWidths.reduce((s, w) => s + w, 0) +
   313→   313→        slashGroupWidth * numSlashes;
   314→   314→
   315→   315→      // Compute start X based on alignment within the optional width
   316→   316→      let startX = x;
   317→   317→      if (options.align === "center" && options.width) {
   318→   318→        startX = x + (options.width - totalWidth) / 2;
   319→   319→      } else if (options.align === "right" && options.width) {
   320→   320→        startX = x + options.width - totalWidth;
   321→   321→      } else if (options.align === "left" || !options.align) {
   322→   322→        startX = x;
   323→   323→      } else if (options.align === "right" && !options.width) {
   324→   324→        startX = x - totalWidth;
   325→   325→      }
   326→   326→
   327→   327→      // Render pieces in visual RTL order:
   328→   328→      // pieces[0] is the rightmost (first read in Arabic), then "  /  ",
   329→   329→      // then pieces[1] on the left, etc.
   330→   330→      let curX = startX;
   331→   331→
   332→   332→      // pieces[0] is the rightmost (first read in Arabic)
   333→   333→      doc.font(fontToUse).fontSize(fontSize).fillColor(color);
   334→   334→      doc.text(trimmedPieces[0], curX, y, {
   335→   335→        features: ["rtla"],
   336→   336→        align: "left",
   337→   337→        lineBreak: false,
   338→   338→      });
   339→   339→      curX += arabicWidths[0];
   340→   340→
   341→   341→      // Then alternating slash-group + next piece
   342→   342→      for (let i = 1; i < trimmedPieces.length; i++) {
   343→   343→        // "  /  " drawn at y + yOffset to align with Arabic baseline
   344→   344→        doc.font(slashFont).fontSize(fontSize).fillColor(color);
   345→   345→        doc.text(slashWithSpaces, curX, y + yOffset, {
   346→   346→          align: "left",
   347→   347→          lineBreak: false,
   348→   348→        });
   349→   349→        curX += slashGroupWidth;
   350→   350→
   351→   351→        doc.font(fontToUse).fontSize(fontSize).fillColor(color);
   352→   352→        doc.text(trimmedPieces[i], curX, y, {
   353→   353→          features: ["rtla"],
   354→   354→          align: "left",
   355→   355→          lineBreak: false,
   356→   356→        });
   357→   357→        curX += arabicWidths[i];
   358→   358→      }
   359→   359→      return;
   360→   360→    }
   361→   361→
   362→   362→    // ============================================================
   363→   363→    // DEFAULT: single-font rendering (with optional rtla feature)

   365→   365→    // rtla GSUB feature breaks ASCII separators (- \ |) into tofu boxes
   366→   366→    // in Noto Sans Arabic. Detect any ASCII separator in the text and
   367→   367→    // disable rtla for that specific text call — Arabic shaping still
   368→   368→    // works correctly without rtla.

   371→   371→      align: "right",
   372→   372→      features: hasAsciiSep ? [] : ["rtla"],
   373→   373→    };

   375→   375→      delete defaultOptions.features;
   376→   376→      options.features = undefined;
   377→   377→    }

   379→   379→    if (options.fontSize) {
   380→   380→      doc.fontSize(options.fontSize);
   381→   381→    }
   382→   382→
   383→   383→    if (options.color) {
   384→   384→      doc.fillColor(options.color);
   385→   385→    }
   386→   386→
   387→   387→    doc.font(fontToUse).text(text, x, y, { ...defaultOptions, ...options });
   388→   388→  };
   389→   389→
   390→   390→  const drawTextEn = (
   391→   391→    text: string,
   392→   392→    x: number,
   393→   393→    y: number,
   394→   394→    options: DrawOpts = {},

   403→   403→  // ============================================================
   404→   404→  // drawMixedText — renders mixed Arabic + Latin/digit text on a
   405→   405→  // single visual baseline.
   406→   406→  //
   407→   407→  // Why: Noto Sans Arabic does NOT have glyphs for ASCII digits
   408→   408→  // (0-9), the forward slash '/', or other Latin punctuation. They
   409→   409→  // render as tofu boxes. The original code only handled '/' as a
   410→   410→  // special case; this helper generalizes that approach to ANY
   411→   411→  // text containing both Arabic and Latin/digit characters.
   412→   412→  //
   413→   413→  // Strategy:
   414→   414→  //   - Split the text into runs: Arabic runs (rendered with NotoArabic
   415→   415→  //     + rtla feature for proper shaping + RTL ordering) and Latin/digit
   416→   416→  //     runs (rendered with Times-Roman).
   417→   417→  //   - Measure each run with its own font.
   418→   418→  //   - Place runs in VISUAL RTL order: rightmost run first.
   419→   419→  //   - Apply a vertical Y-offset to the Latin runs so their visual
   420→   420→  //     baseline aligns with the Arabic visual baseline (because the
   421→   421→  //     Arabic line box is ~2x taller than the Times line box).
   422→   422→  // ============================================================
   423→   423→  const drawMixedText = (
   424→   424→    text: string,
   425→   425→    x: number,
   426→   426→    y: number,
   427→   427→    options: DrawOpts = {},
   428→   428→  ) => {
   429→   429→    if (!useArabicFont) {
   430→   430→      drawTextEn(text, x, y, options);
   431→   431→      return;
   432→   432→    }
   433→   433→
   434→   434→    // Strip Unicode Cf (Format) characters — LRM (U+200E), RLM (U+200F),
   435→   435→    // ZWJ (U+200D), ZWNJ (U+200C), etc. These are BiDi control marks that
   436→   436→    // are needed by the BiDi algorithm but are NOT visually rendered.
   437→   437→    // PDFKit + Latin fonts (Times-Roman, Amiri-Latin) lack glyphs for
   438→   438→    // these codepoints → they render as tofu □ boxes. Mirrors the bot's
   439→   439→    // `clean_text = ''.join(ch for ch in text if unicodedata.category(ch) != 'Cf')`.
   440→   440→    // eslint-disable-next-line no-control-regex
   441→   441→    const CF_REGEX = /[\u200B-\u200F\u202A-\u202E\u2060-\u2064\uFEFF]/g;
   442→   442→    const cleanedText = String(text).replace(CF_REGEX, "");
   443→   443→    text = cleanedText;
   444→   444→
   445→   445→    const fontSize = options.fontSize || 12;
   446→   446→    const color = options.color || "#000000";
   447→   447→    const weight = options.weight || "regular";
   448→   448→
   449→   449→    // Font selection — by default we use the global Arabic font (Noto Sans
   450→   450→    // Arabic) for Arabic runs and Times-Roman for Latin/digit runs.
   451→   451→    // Caller can pass `useAmiri: true` to override BOTH with Amiri
   452→   452→    // (Amiri-Arabic for Arabic runs, Amiri-Latin for Latin/digit runs).
   453→   453→    // Used only for cell 2 of row 2 (Arabic duration cell) per user request.
   454→   454→    const useAmiri = options.useAmiri === true && amiriAvailable;
   455→   455→    const fontArabic = useAmiri
   456→   456→      ? (weight === "bold" ? fontAmiriArabicBoldPath : fontAmiriArabicRegPath)
   457→   457→      : (weight === "bold" ? fontArBold : fontArReg);
   458→   458→    const fontLatin = useAmiri
   459→   459→      ? (weight === "bold" ? fontAmiriLatinBoldPath : fontAmiriLatinRegPath)
   460→   460→      : (weight === "bold" ? fontEnBold : fontEnReg);
   461→   461→
   462→   462→    // Split text into runs of Arabic vs Latin/digit/punctuation.
   463→   463→    // Arabic range: \u0600-\u06FF (Arabic), \u0750-\u077F (Arabic Supplement),
   464→   464→    // \uFB50-\uFDFF (Arabic Presentation Forms-A), \uFE70-\uFEFF (Forms-B),
   465→   465→    // plus Arabic comma/semicolon \u060C \u061B and space (treated as Arabic).
   466→   466→    const arabicChar = /[\u0600-\u06FF\u0750-\u077F\uFB50-\uFDFF\uFE70-\uFEFF]/;
   467→   467→    const isArabicChar = (ch: string) => arabicChar.test(ch);
   468→   468→
   469→   469→    // Tokenize: each run is either all-Arabic-or-space, or all-non-Arabic.
   470→   470→    type Run = { text: string; isArabic: boolean };
   471→   471→    const runs: Run[] = [];
   472→   472→    let i = 0;
   473→   473→    while (i < text.length) {
   474→   474→      const ch = text[i];
   475→   475→      const thisArabic = isArabicChar(ch) || ch === " ";
   476→   476→      let j = i + 1;
   477→   477→      while (j < text.length) {
   478→   478→        const nextArabic = isArabicChar(text[j]) || text[j] === " ";
   479→   479→        if (nextArabic !== thisArabic) break;
   480→   480→        j++;
   481→   481→      }
   482→   482→      // Trailing space on a Latin run should be moved to the next Arabic run
   483→   483→      // (or treated as Arabic) — for simplicity we keep runs as parsed but
   484→   484→      // later render them with appropriate fonts.
   485→   485→      runs.push({ text: text.slice(i, j), isArabic: thisArabic });
   486→   486→      i = j;
   487→   487→    }
   488→   488→
   489→   489→    // Skip mixed-text rendering if there's only one Arabic run (no digits/Latin).
   490→   490→    // In that case fall back to drawTextAr default.
   491→   491→    const hasMixed = runs.some((r) => r.isArabic) && runs.some((r) => !r.isArabic);
   492→   492→    if (!hasMixed) {
   493→   493→      // All Arabic, or all Latin — fall back to default path.
   494→   494→      // If text is pre-shaped (post-BiDi presentation forms), do NOT apply
   495→   495→      // rtla — it would re-shape already-shaped chars and break them.
   496→   496→      const defaultOptions: DrawOpts = {
   497→   497→        align: "right",
   498→   498→        features: options.preShaped ? [] : ["rtla"],
   499→   499→      };
   500→   500→      if (options.fontSize) doc.fontSize(options.fontSize);
   501→   501→      if (options.color) doc.fillColor(options.color);
   502→   502→      doc.font(fontArabic).text(text, x, y, { ...defaultOptions, ...options });
   503→   503→      return;
   504→   504→    }
   505→   505→
   506→   506→    // Compute run widths with their respective fonts
   507→   507→    doc.font(fontArabic).fontSize(fontSize);
   508→   508→    const arabicLineH = doc.heightOfString("م");
   509→   509→    const arabicRunWidths = runs.map((r) =>
   510→   510→      r.isArabic ? doc.widthOfString(r.text) : 0,
   511→   511→    );
   512→   512→
   513→   513→    doc.font(fontLatin).fontSize(fontSize);
   514→   514→    const latinLineH = doc.heightOfString("0");
   515→   515→    const latinRunWidths = runs.map((r) =>
   516→   516→      !r.isArabic ? doc.widthOfString(r.text) : 0,
   517→   517→    );
   518→   518→
   519→   519→    const runWidths = runs.map((r, idx) =>
   520→   520→      r.isArabic ? arabicRunWidths[idx] : latinRunWidths[idx],
   521→   521→    );
   522→   522→
   523→   523→    const totalWidth = runWidths.reduce((s, w) => s + w, 0);
   524→   524→
   525→   525→    // Compute start X based on alignment
   526→   526→    let startX = x;
   527→   527→    if (options.align === "center" && options.width) {
   528→   528→      startX = x + (options.width - totalWidth) / 2;
   529→   529→    } else if (options.align === "right" && options.width) {
   530→   530→      startX = x + options.width - totalWidth;
   531→   531→    } else if (options.align === "left" || !options.align) {
   532→   532→      startX = x;
   533→   533→    } else if (options.align === "right" && !options.width) {
   534→   534→      startX = x - totalWidth;
   535→   535→    }
   536→   536→
   537→   537→    // ------------------------------------------------------------
   538→   538→    // Vertical centering inside a cell of known height.
   539→   539→    //
   540→   540→    // When `centerVertically: true` is passed (with `cellHeight`),
   541→   541→    // we treat the caller's `y` as the TOP of the cell, compute the
   542→   542→    // total visual block height as max(arabicLineH, latinLineH),
   543→   543→    // and shift `y` DOWN by (cellHeight - blockH) / 2 so the whole
   544→   544→    // line block is centered within the cell — mirroring the
   545→   545→    // Python bot's `self.write(height, char)` vertical distribution.
   546→   546→    //
   547→   547→    // `centerVertically: true` also forces `alignTop` to false so the
   548→   548→    // Latin baseline-offset is applied (Latin runs are pushed DOWN to
   549→   549→    // align with the Arabic baseline). This keeps digits, dashes, and
   550→   550→    // parentheses visually centered on the same line as the Arabic
   551→   551→    // letters — matching the bot's mixed-font cell behavior.
   552→   552→    // ------------------------------------------------------------
   553→   553→    let yRender = y;
   554→   554→    const centerVertically = options.centerVertically === true;
   555→   555→    const cellHeight = typeof options.cellHeight === "number" ? options.cellHeight : 0;
   556→   556→    if (centerVertically && cellHeight > 0) {
   557→   557→      const blockH = Math.max(arabicLineH, latinLineH);
   558→   558→      yRender = y + (cellHeight - blockH) / 2;
   559→   559→    }
   560→   560→
   561→   561→    // Vertical offset: shift Latin runs DOWN to align with Arabic baseline.
   562→   562→    // The default behavior pushes Latin runs down by (arabicLineH - latinLineH)
   563→   563→    // so that Latin baselines visually align with Arabic baselines.
   564→   564→    //
   565→   565→    // `alignTop: true` option DISABLES this offset — all runs render at the
   566→   566→    // SAME Y (top of the line box). This mirrors the Python bot's
   567→   567→    // render_mixed_font_cell_v2, which writes each char at the same Y with
   568→   568→    // no vertical adjustment.
   569→   569→    //
   570→   570→    // `centerVertically: true` FORCES this offset ON (i.e., alignTop=false)
   571→   571→    // because we want Latin chars to share the Arabic baseline when the
   572→   572→    // block is centered as a whole.
   573→   573→    const alignTopEffective =
   574→   574→      options.alignTop === true && !centerVertically;
   575→   575→    const yOffset = alignTopEffective ? 0 : arabicLineH - latinLineH;
   576→   576→
   577→   577→    // Render runs in visual order (left to right in the line).
   578→   578→    //
   579→   579→    // `preShaped: true` means the text has already been processed by
   580→   580→    // processArabicBiDi — Arabic chars are already in presentation form
   581→   581→    // (U+FE70-U+FEFF). We must NOT apply `features: ["rtla"]` because
   582→   582→    // PDFKit would try to re-shape already-shaped chars, breaking them.
   583→   583→    //
   584→   584→    // `preShaped: false` (default, used by the duration cell + footer)
   585→   585→    // means the text is in logical order. We DO apply rtla so PDFKit
   586→   586→    // shapes the Arabic and applies its BiDi algorithm to each Arabic run.
   587→   587→    const arabicFeatures = options.preShaped ? [] : ["rtla"];
   588→   588→    let curX = startX;
   589→   589→    for (let k = 0; k < runs.length; k++) {
   590→   590→      const run = runs[k];
   591→   591→      if (run.isArabic) {
   592→   592→        doc.font(fontArabic).fontSize(fontSize).fillColor(color);
   593→   593→        doc.text(run.text, curX, yRender, {
   594→   594→          features: arabicFeatures,
   595→   595→          align: "left",
   596→   596→          lineBreak: false,
   597→   597→        });
   598→   598→      } else {
   599→   599→        doc.font(fontLatin).fontSize(fontSize).fillColor(color);
   600→   600→        doc.text(run.text, curX, yRender + yOffset, {
   601→   601→          align: "left",
   602→   602→          lineBreak: false,
   603→   603→        });
   604→   604→      }
   605→   605→      curX += runWidths[k];
   606→   606→    }
   607→   607→  };
   608→   608→
   609→   609→  // ============================================================
   610→   610→  // drawMixedTextCharByChar — renders mixed Arabic + Latin/digit text
   611→   611→  // by drawing EACH CHARACTER INDIVIDUALLY at sequential LTR positions.
   612→   612→  //
   613→   613→  // Why this exists (and is now the PREFERRED path for any mixed
   614→   614→  // Arabic+Latin text where reliable visual ordering matters):
   615→   615→  //   PDFKit ALWAYS applies its internal BiDi algorithm to any text
   616→   616→  //   containing Arabic-range characters (U+0600-U+06FF, U+FB50-U+FEFF).
   617→   617→  //   This BiDi pass REVERSES Arabic runs, so logical-order "يوم" gets
   618→   618→  //   visually reversed to "موي" on screen. There is no PDFKit option
   619→   619→  //   to disable BiDi. In some deployment environments (e.g. serverless
   620→   620→  //   builds) the `features: ["rtla"]` GSUB pass also fails to apply
   621→   621→  //   shaping, leaving Arabic letters in disconnected isolated forms.
   622→   622→  //
   623→   623→  //   The only way to guarantee correct visual output across all
   624→   624→  //   environments is to render each character as a SEPARATE doc.text()
   625→   625→  //   call. A single character cannot be reversed by BiDi (there's
   626→   626→  //   nothing to reverse). By positioning each char explicitly at
   627→   627→  //   increasing X coordinates, we force the visual LTR order to match
   628→   628→  //   the logical string order.
   629→   629→  //
   630→   630→  // Input handling (the function accepts RAW LOGICAL text):
   631→   631→  //   - Arabic chars are pre-shaped INSIDE this function via
   632→   632→  //     `arabicReshaper.convertArabic(...)`. Pre-shaping is critical
   633→   633→  //     because rendering each char individually means PDFKit's rtla
   634→   634→  //     GSUB feature cannot determine the correct form based on
   635→   635→  //     context (it only sees one char at a time). The pre-shaper
   636→   636→  //     chooses initial/medial/final/isolated forms based on the
   637→   637→  //     full string context, preserving the cursive appearance.
   638→   638→  //   - The input is in LOGICAL order (NOT BiDi-reversed). After
   639→   639→  //     pre-shaping we walk the string left-to-right and emit each
   640→   640→  //     char at increasing X positions, so visual LTR == logical.
   641→   641→  //
   642→   642→  // Rendering:
   643→   643→  //   - Strip Cf (format) chars (LRM, RLM, etc.) — same as drawMixedText.
   644→   644→  //   - For each character:
   645→   645→  //     - If Arabic (including presentation forms U+FB50-U+FEFF):
   646→   646→  //       render with Arabic font, features:[] (no rtla — already shaped)
   647→   647→  //     - If Latin/digit/space: render with Latin font
   648→   648→  //   - Place chars left-to-right at increasing X positions.
   649→   649→  //   - Apply Latin-baseline yOffset so Latin chars align with Arabic
   650→   650→  //     baseline (same as drawMixedText's alignTop:false behavior).
   651→   651→  //   - Center the whole line within `options.width`.
   652→   652→  //   - When `centerVertically:true` + `cellHeight` are passed, the
   653→   653→  //     whole line block is centered vertically inside the cell,
   654→   654→  //     mirroring the bot's `self.write(height, char)` behavior.
   655→   655→  // ============================================================
   656→   656→  const drawMixedTextCharByChar = (
   657→   657→    text: string,
   658→   658→    x: number,
   659→   659→    y: number,
   660→   660→    options: DrawOpts = {},
   661→   661→  ) => {
   662→   662→    if (!useArabicFont) {
   663→   663→      drawTextEn(text, x, y, options);
   664→   664→      return;
   665→   665→    }
   666→   666→
   667→   667→    // ------------------------------------------------------------
   668→   668→    // Strip Cf chars (LRM, RLM, ZWJ, etc.) AFTER BiDi processing.
   669→   669→    //
   670→   670→    // IMPORTANT: We must NOT strip LRM (U+200E) BEFORE processArabicBiDi
   671→   671→    // because the LRM marks are essential for the BiDi algorithm to
   672→   672→    // preserve LTR runs (like dates "09-02-2026") within an RTL context.
   673→   673→    // If we strip them before BiDi, the algorithm reverses the digit
   674→   674→    // runs too, producing "2026-02-09" instead of "09-02-2026".
   675→   675→    //
   676→   676→    // So the order is:
   677→   677→    //   1. Apply processArabicBiDi to the raw logical text (WITH LRM marks)
   678→   678→    //      → produces visual-order string with LRM marks still embedded.
   679→   679→    //   2. Strip Cf chars (including LRM) from the visual string.
   680→   680→    //   3. Render each char individually with features:[].
   681→   681→    //
   682→   682→    // This mirrors the Python bot's safe_arabic_mixed + render_mixed_font_cell_v2:
   683→   683→    //   - safe_arabic_mixed: reshape + bidi (keeps LRM marks)
   684→   684→    //   - render_mixed_font_cell_v2: strips Cf chars via unicodedata.category != 'Cf'
   685→   685→    //     THEN renders each char with self.write()
   686→   686→    // ------------------------------------------------------------
   687→   687→    const CF_REGEX = /[\u200B-\u200F\u202A-\u202E\u2060-\u2064\uFEFF]/g;
   688→   688→    const rawText = String(text);
   689→   689→
   690→   690→    const fontSize = options.fontSize || 12;
   691→   691→    const color = options.color || "#000000";
   692→   692→    const weight = options.weight || "regular";
   693→   693→
   694→   694→    // ------------------------------------------------------------
   695→   695→    // Font selection — caller can pass `useAmiri: true` to render
   696→   696→    // BOTH the Arabic and the Latin/digit portions with the Amiri
   697→   697→    // font family (Amiri-Arabic for Arabic chars, Amiri-Latin for
   698→   698→    // Latin/digits). Otherwise we fall back to the default Arabic
   699→   699→    // font (Noto Sans Arabic) + Times-Roman.
   700→   700→    // ------------------------------------------------------------
   701→   701→    const useAmiri = options.useAmiri === true && amiriAvailable;
   702→   702→    const fontArabic = useAmiri
   703→   703→      ? (weight === "bold" ? fontAmiriArabicBoldPath : fontAmiriArabicRegPath)
   704→   704→      : (weight === "bold" ? fontArBold : fontArReg);
   705→   705→    const fontLatin = useAmiri
   706→   706→      ? (weight === "bold" ? fontAmiriLatinBoldPath : fontAmiriLatinRegPath)
   707→   707→      : (weight === "bold" ? fontEnBold : fontEnReg);
   708→   708→
   709→   709→    // ------------------------------------------------------------
   710→   710→    // CRITICAL FIX (matches the Python bot's `safe_arabic_mixed` +
   711→   711→    // `render_mixed_font_cell_v2` exactly):
   712→   712→    //
   713→   713→    // 1. Apply `processArabicBiDi` (= arabic-reshaper + bidi-js) to
   714→   714→    //    the cleaned logical-order string. This produces a string in
   715→   715→    //    VISUAL order where:
   716→   716→    //      - Arabic chars are in their presentation forms
   717→   717→    //        (U+FB50-U+FDFF, U+FE70-U+FEFF), already chosen based on
   718→   718→    //        cursive context (initial/medial/final/isolated).
   719→   719→    //      - The whole string is reordered per Unicode BiDi so visual
   720→   720→    //        LTR on screen reads correctly.
   721→   721→    //      - Latin/digits/punctuation are in their correct visual
   722→   722→    //        positions (PDFKit's BiDi is NOT needed and NOT applied
   723→   723→    //        because each char is rendered individually — a single
   724→   724→    //        char cannot be BiDi-reversed).
   725→   725→    //
   726→   726→    // 2. Render EACH CHARACTER as a separate doc.text() call with
   727→   727→    //    `features:[]` (no rtla — chars are already in presentation
   728→   728→    //    form, applying rtla would try to re-shape single chars and
   729→   729→    //    either fail or produce isolated forms). This mirrors the
   730→   730→    //    bot's `for char in text: self.write(height, char)`.
   731→   731→    //
   732→   732→    // This approach is environment-independent: it does NOT rely on
   733→   733→    // PDFKit's internal BiDi or `features:["rtla"]` GSUB pass, both
   734→   734→    // of which have been observed to fail in serverless deployments
   735→   735→    // (Vercel), producing disconnected/reversed Arabic glyphs.
   736→   736→    // ------------------------------------------------------------
   737→   737→    // Step 1: Apply processArabicBiDi to the raw logical text WITH LRM
   738→   738→    // marks intact. The LRM marks (U+200E) are essential for BiDi to
   739→   739→    // preserve LTR runs (dates, numbers) within the RTL Arabic context.
   740→   740→    // The result is a visual-order string with Arabic chars in their
   741→   741→    // presentation forms (initial/medial/final/isolated).
   742→   742→    const visualText = processArabicBiDi(rawText);
   743→   743→
   744→   744→    // Step 2: Strip Cf chars (LRM, RLM, ZWJ, etc.) from the visual
   745→   745→    // string. These marks guided BiDi but have no visual glyph — if
   746→   746→    // left in, they'd render as tofu boxes with Latin fonts.
   747→   747→    const shapedText = visualText.replace(CF_REGEX, "");
   748→   748→
   749→   749→    // Classify chars: Arabic (incl. presentation forms) vs Latin/digit/punct
   750→   750→    // Arabic ranges: U+0600-U+06FF, U+0750-U+077F, U+FB50-U+FDFF, U+FE70-U+FEFF
   751→   751→    const isArabicChar = (ch: string) => {
   752→   752→      const cp = ch.codePointAt(0) || 0;
   753→   753→      return (
   754→   754→        (cp >= 0x0600 && cp <= 0x06ff) ||
   755→   755→        (cp >= 0x0750 && cp <= 0x077f) ||
   756→   756→        (cp >= 0xfb50 && cp <= 0xfdff) ||
   757→   757→        (cp >= 0xfe70 && cp <= 0xfeff)
   758→   758→      );
   759→   759→    };
   760→   760→
   761→   761→    // Measure line heights for baseline alignment
   762→   762→    doc.font(fontArabic).fontSize(fontSize);
   763→   763→    const arabicLineH = doc.heightOfString("م");
   764→   764→    doc.font(fontLatin).fontSize(fontSize);
   765→   765→    const latinLineH = doc.heightOfString("0");
   766→   766→    // yOffset shifts Latin chars DOWN to align baseline with Arabic
   767→   767→    const yOffset = arabicLineH - latinLineH;
   768→   768→
   769→   769→    // Build per-char rendering plan: measure each char's width with its
   770→   770→    // appropriate font, then compute total width and start X for centering.
   771→   771→    type CharPlan = { ch: string; isArabic: boolean; width: number };
   772→   772→    const chars: CharPlan[] = [];
   773→   773→    let totalWidth = 0;
   774→   774→
   775→   775→    for (const ch of shapedText) {
   776→   776→      const isAr = isArabicChar(ch);
   777→   777→      if (isAr) {
   778→   778→        doc.font(fontArabic).fontSize(fontSize);
   779→   779→      } else {
   780→   780→        doc.font(fontLatin).fontSize(fontSize);
   781→   781→      }
   782→   782→      const w = doc.widthOfString(ch);
   783→   783→      chars.push({ ch, isArabic: isAr, width: w });
   784→   784→      totalWidth += w;
   785→   785→    }
   786→   786→
   787→   787→    // Compute start X based on alignment
   788→   788→    let startX = x;
   789→   789→    if (options.align === "center" && options.width) {
   790→   790→      startX = x + (options.width - totalWidth) / 2;
   791→   791→    } else if (options.align === "right" && options.width) {
   792→   792→      startX = x + options.width - totalWidth;
   793→   793→    }
   794→   794→
   795→   795→    // ------------------------------------------------------------
   796→   796→    // Vertical centering inside a cell of known height. Two modes:
   797→   797→    //
   798→   798→    // 1. Default (`centerVertically: true` + `cellHeight`):
   799→   799→    //    Use `blockH = max(arabicLineH, latinLineH)` for centering.
   800→   800→    //    Arabic chars render at `yRender`, Latin chars are pushed DOWN
   801→   801→    //    by `yOffset = arabicLineH - latinLineH` so their baselines
   802→   802→    //    align with the Arabic baseline. The visual block is centered
   803→   803→    //    in the cell — works well when this is the ONLY cell with
   804→   804→    //    mixed text.
   805→   805→    //
   806→   806→    // 2. `alignWithSibling: true` (used for cell 2 of row 2):
   807→   807→    //    The adjacent sibling cell (cell 1) renders plain Latin text
   808→   808→    //    with Times-Roman, centered around `latinLineH`. To make the
   809→   809→    //    digit characters in BOTH cells sit at the SAME visual Y
   810→   810→    //    (i.e. aligned baselines), we center THIS cell's block using
   811→   811→    //    `blockH = latinLineH` (NOT the taller arabicLineH), and we
   812→   812→    //    render Arabic chars HIGHER (at `yRender - yOffset`) so their
   813→   813→    //    baseline still aligns with the Latin baseline.
   814→   814→    //
   815→   815→    //    Net effect: Latin digits in cell 2 align horizontally with
   816→   816→    //    the digits in cell 1; the Arabic words "يوم" and "إلى" sit
   817→   817→    //    visually centered on the same baseline (they extend slightly
   818→   818→    //    higher because Arabic glyphs are taller, which is correct).
   819→   819→    // ------------------------------------------------------------
   820→   820→    let yRender = y;
   821→   821→    let arabicYOffset = 0; // additional Y shift for Arabic chars
   822→   822→    const centerVertically = options.centerVertically === true;
   823→   823→    const alignWithSibling = options.alignWithSibling === true;
   824→   824→    const cellHeight = typeof options.cellHeight === "number" ? options.cellHeight : 0;
   825→   825→    if (centerVertically && cellHeight > 0) {
   826→   826→      if (alignWithSibling) {
   827→   827→        // Center on Latin line height so digits align with sibling cell's digits.
   828→   828→        // Render Arabic chars UP by yOffset to align baselines.
   829→   829→        yRender = y + (cellHeight - latinLineH) / 2;
   830→   830→        arabicYOffset = -yOffset; // negative because Arabic goes UP
   831→   831→      } else {
   832→   832→        // Default: center on max(arabic, latin) height, push Latin DOWN.
   833→   833→        const blockH = Math.max(arabicLineH, latinLineH);

   865→   865→  // renderLongNameCell — port of the bot's render_long_name_cell.
   866→   866→  //
   867→   867→  // Used ONLY for uppercase English name cells (Name + Practitioner Name).
   868→   868→  //
   869→   869→  // Bot behavior (render_long_name_cell, lines 413-482):
   870→   870→  //   1. Measure text width; padding = 4mm each side → available_width.
   871→   871→  //   2. If text fits → single line, centered horizontally AND vertically
   872→   872→  //      via FPDF's cell(width, height, text, align='C') which vertically
   873→   873→  //      centers the text within the cell height.
   874→   874→  //   3. If text doesn't fit → split at word boundaries, putting as many
   875→   875→  //      words as possible on line 1 (majority on top).
   876→   876→  //   4. For 2-line case:
   877→   877→  //        line_height = height / 2  (each line gets HALF the row height)
   878→   878→  //        y_offset = y + (height - line_height*2) / 2 = y  (no extra offset)
   879→   879→  //        Line 1: cell(width, line_height, line1, align='C') at y_offset
   880→   880→  //                → text vertically centered within TOP HALF [y, y+h/2]
   881→   881→  //        Line 2: cell(width, line_height, line2, align='C') at y_offset+line_height
   882→   882→  //                → text vertically centered within BOTTOM HALF [y+h/2, y+h]
   883→   883→  //   5. If no good word-boundary split → render as single line (overflow).
   884→   884→  //
   885→   885→  // Our port:
   886→   886→  //   - Same greedy max-words-on-line-1 word splitting.
   887→   887→  //   - Same single-line vertical centering.
   888→   888→  //   - For 2-line case: each line is centered WITHIN ITS HALF of the row,
   889→   889→  //     matching the bot's per-half centering. Previously we drew line 1
   890→   890→  //     at the TOP of the cell and line 2 at the MIDDLE — which left empty
   891→   891→  //     space at the bottom of each half. Now we compute:
   892→   892→  //       line1Y = cellY + (lineH - naturalLineH) / 2
   893→   893→  //       line2Y = cellY + lineH + (lineH - naturalLineH) / 2
   894→   894→  //     so each line is vertically centered in its respective half.
   895→   895→  //   - No mid-word hyphenation (bot has it as a last-resort fallback for
   896→   896→  //     very long single-word names; we skip it since our cell is narrower
   897→   897→  //     and wrapping at word boundaries is sufficient).
   898→   898→  // ============================================================
   899→   899→  const renderLongNameCell = (
   900→   900→    text: string,
   901→   901→    cellX: number,
   902→   902→    cellY: number,
   903→   903→    cellW: number,
   904→   904→    cellH: number,
   905→   905→    options: DrawOpts = {},
   906→   906→  ) => {
   907→   907→    if (!text) return;
   908→   908→
   909→   909→    const fontSize = options.fontSize || 13;
   910→   910→    const color = options.color || "#000000";
   911→   911→    const weight = options.weight || "regular";
   912→   912→    const fontToUse = weight === "bold" ? fontEnBold : fontEnReg;
   913→   913→    // The caller already passes cellW = subColW - 30, which bakes in 15pt
   914→   914→    // of padding on each side of the sub-column. We DON'T add the bot's
   915→   915→    // extra 4mm padding here — our cell is already ~15pt narrower than
   916→   916→    // the bot's per-side, and adding more padding would push borderline
   917→   917→    // names past the wrap threshold unnecessarily.
   918→   918→    const padding = 0;
   919→   919→    const availableWidth = cellW - padding * 2;
   920→   920→

   922→   922→    doc.font(fontToUse).fontSize(fontSize).fillColor(color);
   923→   923→    const textWidth = doc.widthOfString(text);
   924→   924→    const singleLineH = doc.heightOfString("X", { width: cellW });
   925→   925→
   926→   926→    // Case 1: text fits on a single line → render centered, no wrap
   927→   927→    if (textWidth <= availableWidth) {
   928→   928→      const vy = cellY + (cellH - singleLineH) / 2;
   929→   929→      doc.font(fontToUse).fontSize(fontSize).fillColor(color);
   930→   930→      doc.text(text, cellX, vy, {

   932→   932→        align: "center",
   933→   933→        lineBreak: false,
   934→   934→      });
   935→   935→      return;
   936→   936→    }
   937→   937→
   938→   938→    // Case 2: text too long → find the largest prefix of words that fits
   939→   939→    const words = text.split(" ");
   940→   940→    let line1 = "";
   941→   941→    let line2 = "";
   942→   942→    let foundSplit = false;
   943→   943→
   944→   944→    // Iterate from the largest prefix down to 1 word
   945→   945→    for (let i = words.length; i > 0; i--) {
   946→   946→      const testLine = words.slice(0, i).join(" ");
   947→   947→      if (doc.widthOfString(testLine) <= availableWidth) {
   948→   948→        line1 = testLine;
   949→   949→        line2 = words.slice(i).join(" ");
   950→   950→        foundSplit = true;
   951→   951→        break;
   952→   952→      }
   953→   953→    }
   954→   954→
   955→   955→    // Case 3: no good split → render as single line (force, no wrap)
   956→   956→    if (!foundSplit || !line2) {
   957→   957→      const vy = cellY + (cellH - singleLineH) / 2;
   958→   958→      doc.font(fontToUse).fontSize(fontSize).fillColor(color);
   959→   959→      doc.text(text, cellX, vy, {
   960→   960→        width: cellW,
   961→   961→        align: "center",
   962→   962→        lineBreak: false,
   963→   963→      });
   964→   964→      return;
   965→   965→    }
   966→   966→
   967→   967→    // Case 4: render 2 lines — both lines clustered together in the
   968→   968→    // vertical center of the row, with a small gap between them.
   969→   969→    //
   970→   970→    // User explicitly requested a SMALLER gap between the two lines
   971→   971→    // (originally we used bot's per-half centering which left a large
   972→   972→    // gap because each line sat in the center of its half-row).
   973→   973→    //
   974→   974→    // New approach:
   975→   975→    //   - Use natural single-line height for both lines.
   976→   976→    //   - Add a small inter-line gap (4pt) — visually tighter than half-row.
   977→   977→    //   - Cluster both lines as a single block, centered vertically in
   978→   978→    //     the entire row height.
   979→   979→    //
   980→   980→    // Computation:
   981→   981→    //   totalBlockH = singleLineH * 2 + gap
   982→   982→    //   blockTopY   = cellY + (cellH - totalBlockH) / 2
   983→   983→    //   line1Y      = blockTopY
   984→   984→    //   line2Y      = blockTopY + singleLineH + gap
   985→   985→    const gap = 4; // tight inter-line gap (pt)
   986→   986→    const totalBlockH = singleLineH * 2 + gap;
   987→   987→    const blockTopY = cellY + (cellH - totalBlockH) / 2;
   988→   988→    const line1Y = blockTopY;
   989→   989→    const line2Y = blockTopY + singleLineH + gap;
   990→   990→
   991→   991→    doc.font(fontToUse).fontSize(fontSize).fillColor(color);
   992→   992→    doc.text(line1, cellX, line1Y, {
   993→   993→      width: cellW,
   994→   994→      align: "center",
   995→   995→      lineBreak: false,
   996→   996→    });
   997→   997→    doc.text(line2, cellX, line2Y, {
   998→   998→      width: cellW,
   999→   999→      align: "center",
  1000→  1000→      lineBreak: false,
  1001→  1001→    });
  1002→  1002→  };

  1645→  1645→      //     `${LRM}رقم الترخيص : ${licNum}`
  1646→  1646→      //
  1647→  1647→      // Why this order (Arabic label first, digits last):
  1648→  1648→      //   The user reported that the previous layout ("numbers before
  1649→  1649→      //   text") was wrong and asked for the reading order to be:
  1650→  1650→      //       text → colon → numbers
  1651→  1651→      //   In Arabic RTL reading, "text first" means the text is on the
  1652→  1652→      //   RIGHT (where RTL reading starts), and "numbers last" means the
  1653→  1653→      //   numbers are on the LEFT. So the visual LTR layout must be:
  1654→  1654→      //       "1410101201200443 : رقم الترخيص"
  1655→  1655→      //   (digits on the LEFT, colon in the middle, Arabic on the RIGHT)
  1656→  1656→      //
  1657→  1657→      //   bidi-js treats any paragraph starting with an Arabic char (or
  1658→  1658→      //   a sequence whose first strong char is Arabic) as RTL base
  1659→  1659→      //   direction. With RTL base, runs are placed right-to-left in
  1660→  1660→      //   logical order, so:
  1661→  1661→      //     - The Arabic run (first in logical) goes to the RIGHT side.
  1662→  1662→      //     - The colon (next) goes to the MIDDLE.
  1663→  1663→      //     - The digits run (last in logical) goes to the LEFT side.
  1664→  1664→      //   This produces exactly the desired visual layout.
  1665→  1665→      //
  1666→  1666→      //   The LRM (U+200E) prefix is defensive: bidi-js's paragraph-level
  1667→  1667→      //   detection appears to ignore it (both with and without LRM, the
  1668→  1668→      //   visual output is identical), but keeping it makes the intent
  1669→  1669→      //   explicit and protects against any future BiDi implementation
  1670→  1670→      //   that might honor LRM as a base-direction hint. The LRM is
  1671→  1671→      //   stripped automatically by drawMixedTextCharByChar (Cf char)
  1672→  1672→      //   AFTER BiDi processing, so it never renders as tofu.
  1673→  1673→      //
  1674→  1674→      //   The PREVIOUS layout used `${licNum} : رقم الترخيص` (digits
  1675→  1675→      //   first in logical), which produced the OPPOSITE visual:
  1676→  1676→      //   Arabic on LEFT, digits on RIGHT — read in RTL as
  1677→  1677→      //   "digits → colon → text" (numbers before text), which is what
  1678→  1678→      //   the user objected to.
  1679→  1679→      //
  1680→  1680→      // Why char-by-char (not drawMixedText):
  1681→  1681→      //   PDFKit's internal BiDi + `features:["rtla"]` path works in local
  1682→  1682→      //   tests but has been observed to FAIL in some production
  1683→  1683→      //   environments, producing garbled Arabic ("ص.يخزتلا م.ق.ر" instead
  1684→  1684→      //   of "رقم الترخيص") when the rtla GSUB pass doesn't apply shaping.
  1685→  1685→      //   drawMixedTextCharByChar pre-shapes the Arabic internally (via
  1686→  1686→      //   arabic-reshaper) and renders each char as a separate doc.text()
  1687→  1687→      //   call with `features:[]`, bypassing both BiDi reversal and rtla
  1688→  1688→      //   GSUB — so the visual output is identical in every environment.
  1689→  1689→      const LRM_LIC = "\u200e";
  1690→  1690→      const fullLine = `${LRM_LIC}رقم الترخيص : ${licNum}`;
  1691→  1691→
  1692→  1692→      drawMixedTextCharByChar(fullLine, rightCenterX - 125, footerY + 165, {
  1693→  1693→        width: 250,
  1694→  1694→        align: "center",
  1695→  1695→        weight: "bold",
  1696→  1696→        fontSize: 12,
  1697→  1697→        color: "#000000",
  1698→  1698→      });
  1699→  1699→    }
  1700→  1700→  }
  1701→  1701→
  1702→  1702→  // ============================================================
  1703→  1703→  // BOTTOM FOOTER — time + date + national info logo
  1704→  1704→  // ============================================================
  1705→  1705→
  1706→  1706→  const bottomY = pageHeight - 150;
  1707→  1707→  const now = new Date();
  1708→  1708→  const dateStr = now.toLocaleDateString("en-US", {
  1709→  1709→    weekday: "long",
  1710→  1710→    year: "numeric",
  1711→  1711→    month: "long",
  1712→  1712→    day: "numeric",
  1713→  1713→  });
  1714→  1714→  const timeStr = now.toLocaleTimeString("en-US", {
  1715→  1715→    hour: "2-digit",
  1716→  1716→    minute: "2-digit",
  1717→  1717→  });
  1718→  1718→
  1719→  1719→  doc.font(fontEnBold).fontSize(12).fillColor("#000000");
  1720→  1720→  doc.text(timeStr, 40, bottomY);
  1721→  1721→  doc.text(dateStr, 40, bottomY + 20);
  1722→  1722→
  1723→  1723→  if (fs.existsSync(nationalLogo)) {
  1724→  1724→    doc.image(nationalLogo, pageWidth - 160, bottomY - 20, { width: 120 });
  1725→  1725→  }
  1726→  1726→
  1727→  1727→  doc.end();
  1728→  1728→
  1729→  1729→  // Wait for the 'end' event before resolving
  1730→  1730→  return new Promise<Buffer>((resolve, reject) => {
  1731→  1731→    doc.on("end", () => resolve(Buffer.concat(chunks)));
  1732→  1732→    doc.on("error", reject);
  1733→  1733→  });
