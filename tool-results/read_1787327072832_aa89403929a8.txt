   411→    formatGregorian(dateString) {
   412→        if(!dateString) return "";
   413→        const parts = dateString.split('-');
   414→        if(parts.length===3) return `${parts[2]}-${parts[1]}-${parts[0]}`;

   418→    formatAMPM(timeStr) {
   419→        if(!timeStr) return "";
   420→        let [hours, minutes] = timeStr.split(':');
   421→        hours = parseInt(hours);
   422→        let ampm = hours >= 12 ? 'PM' : 'AM';
   423→        hours = hours % 12;
   424→        hours = hours ? hours : 12; 
   425→        return `${hours}:${minutes} ${ampm}`;
   426→    },

   905→    downloadLastPdf() {
   906→        if (!app.state.lastPdfBase64) {
   907→            if(this.tg) this.tg.showAlert('لا يوجد تقرير للتحميل');
   908→            else alert('لا يوجد تقرير للتحميل');
   909→            return;
   910→        }
   911→        try {
   912→            const link = document.createElement('a');
   913→            link.href = app.state.lastPdfBase64;
   914→            link.download = `${app.state.lastReportId || 'report'}.pdf`;
   915→            document.body.appendChild(link);
   916→            link.click();
   917→            document.body.removeChild(link);
   918→        } catch(e) {
   919→            if(this.tg) this.tg.showAlert('حدث خطأ أثناء التحميل');
   920→            else alert('حدث خطأ أثناء التحميل');
   921→        }
   922→    },

   924→    printLastPdf() {
   925→        if (!app.state.lastPdfBase64) {
   926→            if(this.tg) this.tg.showAlert('لا يوجد تقرير للطباعة');
   927→            else alert('لا يوجد تقرير للطباعة');
   928→            return;
   929→        }
   930→        try {
   931→            const rawBase64 = app.state.lastPdfBase64.split(',')[1] || app.state.lastPdfBase64;
   932→            const binaryStr = atob(rawBase64);
   933→            const bytes = new Uint8Array(binaryStr.length);
   934→            for (let i = 0; i < binaryStr.length; i++) {
   935→                bytes[i] = binaryStr.charCodeAt(i);
   936→            }
   937→            const blob = new Blob([bytes], { type: 'application/pdf' });
   938→            const blobUrl = URL.createObjectURL(blob);
   939→
   940→            // Use hidden iframe for reliable cross-browser printing
   941→            const iframe = document.createElement('iframe');
   942→            iframe.style.position = 'fixed';
   943→            iframe.style.right = '0';
   944→            iframe.style.bottom = '0';
   945→            iframe.style.width = '0';
   946→            iframe.style.height = '0';
   947→            iframe.style.border = 'none';
   948→            iframe.style.opacity = '0';
   949→            iframe.style.pointerEvents = 'none';
   950→            document.body.appendChild(iframe);
   951→
   952→            iframe.onload = function() {
   953→                setTimeout(() => {
   954→                    try {
   955→                        iframe.contentWindow.print();
   956→                    } catch(e) {
   957→                        console.warn('iframe print failed, trying window.open:', e);
   958→                        // Fallback to new window
   959→                        const printWindow = window.open(blobUrl, '_blank');
   960→                        if (printWindow) {
   961→                            printWindow.addEventListener('load', function() {
   962→                                setTimeout(() => { printWindow.print(); }, 800);
   963→                            });
   964→                        } else {
   965→                            app.downloadLastPdf();
   966→                        }
   967→                    }
   968→                    // Cleanup iframe after printing
   969→                    setTimeout(() => {
   970→                        document.body.removeChild(iframe);
   971→                        URL.revokeObjectURL(blobUrl);
   972→                    }, 5000);
   973→                }, 500);
   974→            };
   975→            iframe.src = blobUrl;
   976→        } catch(e) {
   977→            console.error('Print failed:', e);
   978→            app.downloadLastPdf();
   979→        }
   980→    },
   981→

   986→            window.close();
   987→        }
