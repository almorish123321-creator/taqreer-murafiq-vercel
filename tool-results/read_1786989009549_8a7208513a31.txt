   709→@router.post("", response_model=ThreadResponse)
   710→async def create_thread(body: ThreadCreateRequest, request: Request) -> ThreadResponse:
   711→    """Create a new thread.
   712→
   713→    Writes a thread_meta record (so the thread appears in /threads/search)
   714→    and an empty checkpoint (so state endpoints work immediately).
   715→    Idempotent: returns the existing record when ``thread_id`` already exists.
   716→    """
   717→    from app.gateway.deps import get_thread_store
   718→
   719→    checkpointer = get_checkpointer(request)
   720→    thread_store = get_thread_store(request)
   721→    thread_id = resolve_thread_id(body.thread_id)
   722→    now = now_iso()
   723→    thread_owner_user_id = get_trusted_internal_owner_user_id(request)
   724→    thread_owner_kwargs = {"user_id": thread_owner_user_id} if thread_owner_user_id else {}
   725→    # ``body.metadata`` is already stripped of server-reserved keys by
   726→    # ``ThreadCreateRequest._strip_reserved`` — see the model definition.
   727→
   728→    # Idempotency: return existing record when already present
   729→    existing_record = await _resolve_existing_thread(thread_store, thread_id, thread_owner_user_id, thread_owner_kwargs)
   730→    if existing_record is not None:
   731→        return _existing_thread_response(thread_id, existing_record)
   732→
   733→    # Write thread_meta so the thread appears in /threads/search immediately
   734→    try:
   735→        await thread_store.create(
   736→            thread_id,
   737→            assistant_id=getattr(body, "assistant_id", None),
   738→            **thread_owner_kwargs,
   739→            metadata=body.metadata,
   740→        )
   741→    except IntegrityError:
   742→        # The idempotency read above and this insert are not atomic: a
   743→        # concurrent request for the same thread_id can commit in between, so
   744→        # the SQL-backed store rejects ours on the duplicate primary key.
   745→        # Honour the documented idempotency contract by resolving the
   746→        # now-existing record — running the same owner reconciliation the fast
   747→        # path does — instead of surfacing the conflict as a 500. (The memory
   748→        # store overwrites rather than raising, so it never reaches here.)
   749→        existing_record = await _resolve_existing_thread(thread_store, thread_id, thread_owner_user_id, thread_owner_kwargs)
   750→        if existing_record is not None:
   751→            return _existing_thread_response(thread_id, existing_record)
   752→        # A duplicate-key error with no row we can read back is a real failure.
   753→        logger.exception("Failed to write thread_meta for %s", sanitize_log_param(thread_id))
   754→        raise HTTPException(status_code=500, detail="Failed to create thread")
   755→    except Exception:
   756→        # Any non-race failure must surface, not be silently swallowed as a 200.
   757→        logger.exception("Failed to write thread_meta for %s", sanitize_log_param(thread_id))
   758→        raise HTTPException(status_code=500, detail="Failed to create thread")
   759→
   760→    # Write an empty checkpoint so state endpoints work immediately
   761→    config = {"configurable": {"thread_id": thread_id, "checkpoint_ns": ""}}
   762→    try:
   763→        ckpt_metadata = {
   764→            "step": -1,
   765→            "source": "input",
   766→            "writes": None,
   767→            "parents": {},
   768→            **body.metadata,
   769→            "created_at": now,
   770→        }
   771→        await checkpointer.aput(config, empty_checkpoint(), ckpt_metadata, {})
   772→    except Exception:
   773→        logger.exception("Failed to create checkpoint for thread %s", sanitize_log_param(thread_id))
   774→        raise HTTPException(status_code=500, detail="Failed to create thread")
   775→
   776→    logger.info("Thread created: %s", sanitize_log_param(thread_id))
   777→    return ThreadResponse(
   778→        thread_id=thread_id,
   779→        status="idle",
   780→        created_at=now,
   781→        updated_at=now,
   782→        metadata=body.metadata,
   783→    )

   786→@router.post("/{thread_id}/branches", response_model=ThreadBranchResponse)
   787→@require_permission("threads", "write", owner_check=True, require_existing=True)
   788→async def branch_thread(thread_id: ThreadId, body: ThreadBranchRequest, request: Request) -> ThreadBranchResponse:
   789→    """Create a new main-thread branch from a completed assistant turn."""
   790→    from app.gateway.deps import get_thread_store
   791→
   792→    thread_store = get_thread_store(request)
   793→
   794→    source_record = await thread_store.get(thread_id)
   795→    if source_record is None:
   796→        raise HTTPException(status_code=404, detail=f"Thread {thread_id} not found")
   797→
   798→    source_metadata = source_record.get("metadata") or {}
   799→    if source_metadata.get(_SIDECAR_METADATA_KEY) is True:
   800→        raise HTTPException(status_code=409, detail="Branching is only available in the main conversation.")
   801→    source_accessor, source_config = build_checkpoint_state_accessor(
   802→        request,
   803→        thread_id=thread_id,
   804→        assistant_id=source_record.get("assistant_id"),
   805→    )
   806→
   807→    target_message_ids = {body.message_id, *body.message_ids}
   808→    snapshot = await _find_branch_checkpoint(source_accessor, source_config, target_message_ids)
   809→    parent_checkpoint_id = _checkpoint_id(snapshot)
   810→    if not parent_checkpoint_id:
   811→        raise HTTPException(status_code=409, detail="This turn can no longer be branched from.")
   812→    target_human = _branch_target_human_message(_checkpoint_messages(snapshot), target_message_ids)
   813→    target_human_id = _message_id(target_human)
   814→    if not target_human_id:
   815→        raise HTTPException(status_code=409, detail="This turn can no longer be branched from.")
   816→    replay_base_tuple = await _find_branch_replay_base(
   817→        source_accessor,
   818→        source_config,
   819→        snapshot,
   820→        target_human_id,
   821→    )
   822→
   823→    # Workspace files are not checkpointed, so they only reflect the *current* thread
   824→    # state. Cloning them onto a branch from an older turn would leak files created
   825→    # after that turn (message history rolls back, workspace would not). Restrict the
   826→    # best-effort clone to branches taken from the latest turn so history and workspace
   827→    # stay consistent.
   828→    branch_from_latest_turn = await _branch_targets_latest_turn(source_accessor, source_config, target_message_ids)
   829→
   830→    new_thread_id = str(uuid.uuid4())
   831→    now = now_iso()
   832→    branch_metadata = {
   833→        _BRANCH_METADATA_KEY: True,
   834→        "branch_parent_thread_id": thread_id,
   835→        "branch_parent_checkpoint_id": parent_checkpoint_id,
   836→        "branch_parent_message_id": body.message_id,
   837→        "branch_created_at": now,
   838→    }
   839→
   840→    display_name = body.title or _default_branch_display_name(
   841→        source_record.get("display_name"),
   842→        source_is_branch=source_metadata.get(_BRANCH_METADATA_KEY) is True,
   843→    )
   844→    thread_owner_user_id = get_trusted_internal_owner_user_id(request)
   845→    thread_owner_kwargs = {"user_id": thread_owner_user_id} if thread_owner_user_id else {}
   846→
   847→    # Copy materialized values with replace semantics: reducer channels must
   848→    # not re-merge an already-aggregated value, so every copied reducer value
   849→    # is wrapped in Overwrite (not just messages).
   850→    branch_accessor, new_config = build_checkpoint_state_mutation_accessor(
   851→        request,
   852→        thread_id=new_thread_id,
   853→        as_node="branch",
   854→        # The branch write carries the full materialized snapshot; use the
   855→        # source assistant's effective schema so extension middleware channels
   856→        # survive instead of being silently discarded as unknown channels.
   857→        state_schema=graph_state_schema(getattr(source_accessor, "graph", None)),
   858→    )
   859→    branch_reducer_fields = graph_reducer_channels(getattr(branch_accessor, "graph", None))
   860→    if branch_reducer_fields is None:
   861→        branch_reducer_fields = THREAD_STATE_REDUCER_FIELDS
   862→
   863→    def branch_values(source_snapshot: Any) -> dict[str, Any]:
   864→        values: dict[str, Any] = {}
   865→        for key, value in dict(source_snapshot.values).items():
   866→            if key in _BRANCH_EXCLUDED_CHANNELS:
   867→                continue
   868→            if key in branch_reducer_fields:
   869→                values[key] = Overwrite(list(value) if key == "messages" and isinstance(value, list) else value)
   870→            else:
   871→                values[key] = value
   872→        return values
   873→
   874→    # Stamp both synthetic checkpoints with the branch-creation time because
   875→    # serializers fall back to metadata when snapshot.created_at is absent.
   876→    checkpoint_metadata_updates = {
   877→        **branch_metadata,
   878→        "source": "branch",
   879→        "updated_at": now,
   880→        "created_at": now,
   881→    }
   882→    new_config.setdefault("metadata", {}).update(checkpoint_metadata_updates)
   883→    try:
   884→        head_config = new_config
   885→        if replay_base_tuple is not None:
   886→            head_config = await branch_accessor.aupdate(
   887→                new_config,
   888→                branch_values(replay_base_tuple),
   889→                as_node="branch",
   890→            )
   891→            head_config.setdefault("metadata", {}).update(checkpoint_metadata_updates)
   892→        await branch_accessor.aupdate(
   893→            head_config,
   894→            branch_values(snapshot),
   895→            as_node="branch",
   896→        )
   897→    except _CHECKPOINT_MODE_ERRORS as exc:
   898→        raise _checkpoint_mode_http_error(exc, new_thread_id) from exc
   899→    except Exception:
   900→        logger.exception("Failed to write branch checkpoint for thread %s", sanitize_log_param(new_thread_id))
   901→        raise HTTPException(status_code=500, detail="Failed to create branch") from None
   902→
   903→    try:
   904→        await thread_store.create(
   905→            new_thread_id,
   906→            assistant_id=source_record.get("assistant_id"),
   907→            display_name=display_name,
   908→            metadata=branch_metadata,
   909→            **thread_owner_kwargs,
   910→        )
   911→    except Exception:
   912→        logger.exception("Failed to write branch thread_meta for %s", sanitize_log_param(new_thread_id))
   913→        raise HTTPException(status_code=500, detail="Failed to create branch") from None
   914→
   915→    # The thread feed (GET /messages, /messages/page) reads the run-event
   916→    # store, not checkpoints, and a fresh branch has no run_events — so the
   917→    # inherited history would vanish from the UI as soon as the branch's
   918→    # first run refreshes the feed (#4380 problem 2). Seed the branch's
   919→    # run_events from the same checkpoint snapshot the branch was created
   920→    # from. Best-effort: on failure the branch stays usable, with history
   921→    # visible only through the checkpoint overlay until it is re-branched.
   922→    try:
   923→        seed_events = build_branch_history_seed_events(
   924→            _checkpoint_messages(snapshot),
   925→            thread_id=new_thread_id,
   926→            run_id_prefix=f"branch-seed-{new_thread_id}",
   927→            parent_thread_id=thread_id,
   928→        )
   929→        if seed_events:
   930→            await get_run_event_store(request).put_batch(seed_events)
   931→            history_seed_mode = "seeded"
   932→        else:
   933→            history_seed_mode = "skipped_empty"
   934→    except Exception:
   935→        logger.exception("Failed to seed branch history run-events for thread %s", sanitize_log_param(new_thread_id))
   936→        history_seed_mode = "failed"
   937→
   938→    if branch_from_latest_turn:
   939→        workspace_clone_mode = await _copy_branch_user_data(thread_id, new_thread_id)
   940→    else:
   941→        workspace_clone_mode = "skipped_historical_turn"
   942→    return ThreadBranchResponse(
   943→        thread_id=new_thread_id,
   944→        parent_thread_id=thread_id,
   945→        parent_checkpoint_id=parent_checkpoint_id,
   946→        branched_from_message_id=body.message_id,
   947→        workspace_clone_mode=workspace_clone_mode,
   948→        history_seed_mode=history_seed_mode,
   949→    )

  1336→    thread_id: ThreadId,
  1337→    body: ThreadHistoryRequest,
  1338→    request: Request,
  1339→    background_tasks: BackgroundTasks,
  1340→) -> list[HistoryEntry]:
  1341→    """Get materialized graph state history for a thread.
  1342→
  1343→    Only the latest (first) checkpoint carries the ``messages`` key to
  1344→    avoid duplicating the complete conversation across every entry.
  1345→    """

  1356→    entries: list[HistoryEntry] = []

  1412→                        # event-store + run-manager, then upgrade by a

  1433→                                        msg_to_run[content["id"]] = run_id
  1434→

  1436→                                for msg in serialized_msgs:

  1438→                                        additional_kwargs = msg.get("additional_kwargs")
